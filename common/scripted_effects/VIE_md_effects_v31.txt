# VPA_Land_Readiness: 1 -> 2 -> 3 -> 4
VIE_v31_upgrade_readiness = {
	if = {
		limit = { has_idea = VPA_Land_Readiness_3 }
		swap_ideas = {
			remove_idea = VPA_Land_Readiness_3
			add_idea = VPA_Land_Readiness_4
		}
	}
	else_if = {
		limit = { has_idea = VPA_Land_Readiness_2 }
		swap_ideas = {
			remove_idea = VPA_Land_Readiness_2
			add_idea = VPA_Land_Readiness_3
		}
	}
	else_if = {
		limit = { has_idea = VPA_Land_Readiness_1 }
		swap_ideas = {
			remove_idea = VPA_Land_Readiness_1
			add_idea = VPA_Land_Readiness_2
		}
	}
	else = {
		add_ideas = VPA_Land_Readiness_1
	}
}

VIE_v31_check_synergy = {
	if = {
		limit = {
			has_completed_focus = VIE_lf_sf_special_warfare_capstone
		}
		if = {
			limit = {
				has_completed_focus = VIE_lf_operational_counteroffensive_capstone
				NOT = { has_idea = VIE_synergy_mech_commando_idea }
			}
			add_ideas = VIE_synergy_mech_commando_idea
		}
		if = {
			limit = {
				has_completed_focus = VIE_lf_complex_terrain_mobile_capstone
				NOT = { has_idea = VIE_synergy_mobility_commando_idea }
			}
			add_ideas = VIE_synergy_mobility_commando_idea
		}
	}
}
