			name = "Lu doan Dac cong Tinh nhue"
			regiments = {
				Special_Forces = { x = 0 y = 0 }
				Special_Forces = { x = 0 y = 1 }

				Special_Forces = { x = 1 y = 0 }
				Special_Forces = { x = 1 y = 1 }
				Special_Forces = { x = 1 y = 2 }
			}
			support = {
				L_Engi_Comp = { x = 0 y = 0 }
			}
			priority = 2
		}
	}
	capital_scope = {
		create_unit = {
			division = "name = \"Lu doan Dac cong Tinh nhue 1\" division_template = \"Lu doan Dac cong Tinh nhue\" start_experience_factor = 0.7 start_equipment_factor = 0.8"
			owner = ROOT
			count = 1
		}
	}
}
