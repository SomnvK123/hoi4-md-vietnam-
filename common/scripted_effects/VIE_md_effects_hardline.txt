# Vietnam submod scripted effects: "Con duong Kien dinh" (CPV Hardline, ruling party 4).
# Reform pressure (VIE_rp, 0-100) is the only new variable of the branch:
#   0-30 stable | 31-55 simmering | 56-79 tense (idea) | 80-100 crisis (idea, forced event)
# Callers add pressure through a temp variable:
#   set_temp_variable = { VIE_rp_add = 6 }  VIE_hl_add_pressure = yes
# No parametrized effects here, so file load order does not matter.

# Adds VIE_rp_add (temp, may be negative) to the reform pressure. A rise starts the 120 day "recently tightened"
# window during which the monthly decay is paused.
VIE_hl_add_pressure = {
	hidden_effect = { VIE_hl_add_pressure_inner = yes }
}
VIE_hl_add_pressure_inner = {
	add_to_variable = { var = VIE_rp value = VIE_rp_add }
	clamp_variable = { var = VIE_rp min = 0 max = 100 }
	# During the self-reliance period (event vie_hl.7, "persist") the pressure stops at 80
	if = {
		limit = {
			has_idea = VIE_hl_self_reliance_idea
			check_variable = { VIE_rp > 80 }
		}
		set_variable = { VIE_rp = 80 }
	}
	if = {
		limit = { check_variable = { VIE_rp_add > 0 } }
		hidden_effect = { set_country_flag = { flag = VIE_hl_rp_recent value = 1 days = 120 } }
	}
	VIE_hl_update_pressure_idea = yes
}

# Keeps exactly the idea that matches the current level: none up to 55, tense 56-79, crisis from 80.
VIE_hl_update_pressure_idea = {
	hidden_effect = { VIE_hl_update_pressure_idea_inner = yes }
}
VIE_hl_update_pressure_idea_inner = {
	if = {
		limit = { check_variable = { VIE_rp > 79 } }
		if = {
			limit = { has_idea = VIE_hl_pressure_tense_idea }
			remove_ideas = VIE_hl_pressure_tense_idea
		}
		if = {
			limit = { NOT = { has_idea = VIE_hl_pressure_crisis_idea } }
			add_ideas = VIE_hl_pressure_crisis_idea
		}
	}
	else_if = {
		limit = { check_variable = { VIE_rp > 55 } }
		if = {
			limit = { has_idea = VIE_hl_pressure_crisis_idea }
			remove_ideas = VIE_hl_pressure_crisis_idea
		}
		if = {
			limit = { NOT = { has_idea = VIE_hl_pressure_tense_idea } }
			add_ideas = VIE_hl_pressure_tense_idea
		}
	}
	else = {
		if = {
			limit = { has_idea = VIE_hl_pressure_tense_idea }
			remove_ideas = VIE_hl_pressure_tense_idea
		}
		if = {
			limit = { has_idea = VIE_hl_pressure_crisis_idea }
			remove_ideas = VIE_hl_pressure_crisis_idea
		}
	}
}

# Monthly tick (on_monthly, see VIE_md_on_actions.txt). Pressure eases by 1 a month when no tightening focus
# was finished in the last 120 days. Reaction events are added to this effect in phase 4.
VIE_hl_monthly = {
	if = {
		limit = { VIE_hl_in_power = yes }
		if = {
			limit = {
				NOT = { has_country_flag = VIE_hl_rp_recent }
				check_variable = { VIE_rp > 0 }
			}
			subtract_from_variable = { VIE_rp = 1 }
		}
		VIE_hl_update_pressure_idea = yes
		VIE_hl_event_scheduler = yes
	}
}

# Reaction events vie_hl.2 - .8 (see events/VIE_md_hardline.txt). Soft events respect the shared 45 day pop-up
# cooldown (VIE_popup_cd) and fire once; the crisis (.7) and the re-evaluation (.8) are forced.
VIE_hl_event_scheduler = {
	# .7 legitimacy crisis: pressure 80+, forced (ignores the cooldown), not while already self-reliant
	if = {
		limit = {
			check_variable = { VIE_rp > 79 }
			NOT = { has_idea = VIE_hl_self_reliance_idea }
			NOT = { has_country_flag = VIE_hl_ev7_pending }
		}
		hidden_effect = { set_country_flag = { flag = VIE_hl_ev7_pending value = 1 days = 60 } }
		country_event = { id = vie_hl.7 days = 2 }
	}
	# .8 re-evaluation: the self-reliance idea has run out
	if = {
		limit = {
			has_country_flag = VIE_hl_self_reliance_running
			NOT = { has_idea = VIE_hl_self_reliance_idea }
			NOT = { has_country_flag = VIE_hl_ev8_pending }
		}
		hidden_effect = { set_country_flag = { flag = VIE_hl_ev8_pending value = 1 days = 60 } }
		country_event = { id = vie_hl.8 days = 2 }
	}
	if = {
		limit = { NOT = { has_country_flag = VIE_popup_cd } }
		# .2 letter from the veterans
		if = {
			limit = {
				check_variable = { VIE_rp > 30 }
				has_completed_focus = VIE_hl_party_rectification
				NOT = { has_country_flag = VIE_hl_ev2_done }
			}
			hidden_effect = { set_country_flag = VIE_hl_ev2_done }
			hidden_effect = { set_country_flag = { flag = VIE_popup_cd value = 1 days = 45 } }
			country_event = { id = vie_hl.2 days = 3 random_days = 15 }
		}
		# .3 investors hold back
		else_if = {
			limit = {
				check_variable = { VIE_rp > 39 }
				has_completed_focus = VIE_hl_selective_fdi
				NOT = { has_country_flag = VIE_hl_ev3_done }
			}
			hidden_effect = { set_country_flag = VIE_hl_ev3_done }
			hidden_effect = { set_country_flag = { flag = VIE_popup_cd value = 1 days = 45 } }
			country_event = { id = vie_hl.3 days = 3 random_days = 15 }
		}
		# .4 labour and trade commitments (EVFTA / CPTPP)
		else_if = {
			limit = {
				OR = {
					has_completed_focus = VIE_evfta
					has_completed_focus = VIE_cptpp_member
				}
				has_completed_focus = VIE_hl_cyber_ideology
				NOT = { has_country_flag = VIE_hl_ev4_done }
			}
			hidden_effect = { set_country_flag = VIE_hl_ev4_done }
			hidden_effect = { set_country_flag = { flag = VIE_popup_cd value = 1 days = 45 } }
			country_event = { id = vie_hl.4 days = 3 random_days = 15 }
		}
		# .5 extraordinary Central Committee meeting: first time pressure reaches 56 (also fired by focus E1)
		else_if = {
			limit = {
				check_variable = { VIE_rp > 55 }
				NOT = { has_country_flag = VIE_hl_ev5_cd }
			}
			hidden_effect = { set_country_flag = { flag = VIE_hl_ev5_cd value = 1 days = 730 } }
			hidden_effect = { set_country_flag = { flag = VIE_popup_cd value = 1 days = 45 } }
			country_event = { id = vie_hl.5 days = 3 random_days = 15 }
		}
		# .6 public reaction over the South China Sea
		else_if = {
			limit = {
				VIE_scs_escalated_trigger = yes
				has_completed_focus = VIE_hl_party_diplomacy
				NOT = { has_country_flag = VIE_hl_ev6_done }
			}
			hidden_effect = { set_country_flag = VIE_hl_ev6_done }
			hidden_effect = { set_country_flag = { flag = VIE_popup_cd value = 1 days = 45 } }
			country_event = { id = vie_hl.6 days = 3 random_days = 15 }
		}
	}
}

# Old integration focuses (evfta, cptpp_member, investment_grade, international_financial_centre) act as a
# relief valve while the hardline party rules: pressure -4, and the cadres resent it until the "socialist
# industrialization" tier legitimises opening up. Safe to call in any regime (does nothing outside party 4).
VIE_hl_integration_valve = {
	if = {
		limit = { VIE_hl_in_power = yes }
		set_temp_variable = { VIE_rp_add = -4 }
		VIE_hl_add_pressure = yes
		custom_effect_tooltip = VIE_hl_rp_m4_tt
		if = {
			limit = { NOT = { has_completed_focus = VIE_hl_five_year_plan } }
			set_temp_variable = { temp_opinion = -2 }
			change_communist_cadres_opinion = yes
			VIE_bop_reform_small = yes
		}
	}
}

# Hardline takes power (called by event vie_hl.1). Same shape as the transition in VIE_sec_cyber_control:
# raise the party's popularity, then go through the single regime gateway. Idempotent.
# VIE_transition_regime resets the balance of power to 0, so it is moved back to -0.5 afterwards.
VIE_hl_take_power = {
	if = {
		limit = { NOT = { check_variable = { ruling_party = 4 } } }
		set_temp_variable = { party_index = 4 }
		set_temp_variable = { party_popularity_increase = 0.15 }
		set_temp_variable = { temp_outlook_increase = 0.15 }
		hidden_effect = { change_relative_party_popularity = yes }
		set_temp_variable = { rul_party_temp = 4 }
		VIE_transition_regime = yes
		if = {
			limit = { check_variable = { ruling_party = 4 } }
			# The historical General Secretaries do not carry over into the hardline era
			hidden_effect = { clr_country_flag = VIE_congress_controls_leader }
			hidden_effect = { clr_country_flag = VIE_trong_third_term }
			hidden_effect = { clr_country_flag = VIE_trong_third_term_confirmed }
			hidden_effect = { clr_country_flag = VIE_to_lam_era }
			if = {
				limit = { has_country_flag = VIE_congress9_phieu }
				set_variable = { VIE_gs = 1 }
			}
			else = {
				set_variable = { VIE_gs = 0 }
			}
			if = {
				limit = { has_country_flag = VIE_bop_active }
				add_power_balance_value = { id = VIE_party_balance value = -0.5 }
			}
			set_variable = { VIE_rp = 20 }
			VIE_hl_update_pressure_idea = yes
		}
	}
}

# Called by VIE_transition_regime when the country leaves ruling party 4 (hooked in phase 2). Drops the branch's
# ideas and the pressure. Ideas of the focus tree itself are added to the list when they are created (phase 3).
VIE_hl_cleanup = {
	if = {
		limit = { has_idea = VIE_hl_pressure_tense_idea }
		remove_ideas = VIE_hl_pressure_tense_idea
	}
	if = {
		limit = { has_idea = VIE_hl_pressure_crisis_idea }
		remove_ideas = VIE_hl_pressure_crisis_idea
	}
	# Ideas of the focus tree. The handover ending (X3) sets VIE_hl_keep_legacy and keeps two of them.
	if = { limit = { has_idea = VIE_hl_state_leading_idea } remove_ideas = VIE_hl_state_leading_idea }
	if = { limit = { has_idea = VIE_hl_state_transition_idea } remove_ideas = VIE_hl_state_transition_idea }
