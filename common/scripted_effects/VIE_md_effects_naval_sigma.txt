VIE_naval_ensure_sigma_variant = {
	if = {
		limit = { NOT = { has_country_flag = VIE_naval_sigma_variant_done } }
		set_country_flag = VIE_naval_sigma_variant_done
		create_equipment_variant = {
			name = "Sigma Class"
			type = frigate_hull_3
			parent_version = 0
			role_icon_index = 2
			modules = {
				fixed_ship_engine_slot = module_light_surface_diesel_power_1
				fixed_ship_fire_control_system_slot = module_digital_integrated_fire_control
				fixed_ship_radar_slot = module_radar_3
				fixed_ship_auxillary_slot_2 = empty
				fixed_ship_auxillary_slot_1 = module_naval_missile_mount_quad
				fixed_ship_auxillary_slot = module_ciws_3
				fixed_ship_battery_slot = module_76mm_gun_3
				front_1_custom_slot = module_naval_missile_mount_quad
				rear_1_custom_slot = module_fuel_tank
				fixed_ship_missile_ammo_slot = module_naval_ammo_balanced_2
			}
		}
	}
}

# Cua so 2011-10 -> 2014-12, doi tac Ha Lan (HOL). Het cua so ma pop-up bi chan: ket qua lich
# su = suspended (khong co Sigma).
VIE_naval_sched_sigma = {
	if = {
		limit = {
			NOT = { has_country_flag = VIE_sigma_offered }
			NOT = { has_country_flag = VIE_sigma_missed }
		}
		if = {
			limit = {
				date > 2011.9.30
				date < 2015.1.1
				country_exists = HOL
				NOT = { has_war_with = HOL }
			}
			if = {
				limit = { NOT = { has_country_flag = VIE_popup_cd } }
				set_country_flag = VIE_sigma_offered
				set_country_flag = { flag = VIE_popup_cd value = 1 days = 45 }
				country_event = { id = vie_naval.40 days = 1 random_days = 14 }
			}
			else = {
				set_country_flag = VIE_sigma_gate_seen
			}
		}
		else_if = {
			limit = { date > 2014.12.31 }
			if = {
				limit = { has_country_flag = VIE_sigma_gate_seen }
				set_country_flag = VIE_sigma_offered
				VIE_fb_naval_sigma = yes
			}
			else = {
				set_country_flag = VIE_sigma_missed
			}
		}
	}
}

VIE_fb_naval_sigma = {
	set_country_flag = VIE_sigma_suspended
}

# Nhanh Historical: sau khi chon "tiep tuc dam phan", thang 08/2013 ban .44 (dam phan dat thoa
# thuan 2 tau nhung chua ky), roi suspended. Neu pop-up dang ban thi thu lai; qua 2014-12 thi
# dong im lang.
VIE_naval_sigma_review = {
	if = {
		limit = {
			has_country_flag = VIE_sigma_historical
			NOT = { has_country_flag = VIE_sigma_suspended }
			date > 2013.7.31
		}
		if = {
			limit = { NOT = { has_country_flag = VIE_popup_cd } date < 2015.1.1 }
			set_country_flag = { flag = VIE_popup_cd value = 1 days = 45 }
			country_event = { id = vie_naval.44 days = 1 random_days = 14 }
			set_country_flag = VIE_sigma_suspended
		}
		else_if = {
			limit = { date > 2014.12.31 }
			VIE_fb_naval_sigma = yes
		}
	}
}

# Chi phi Sigma (ty USD) vao bien tam VIE_naval_cost: so tau x 0,33 x he so cau hinh
# (1,0 / 1,15 / 1,3), cong 25% neu ky muon (VIE_sigma_late). Mot cho tinh duy nhat de Funding
# Gate va hop dong khong lech nhau.
VIE_naval_sigma_cost = {
	set_temp_variable = { VIE_naval_cost = VIE_sigma_qty_ordered }
	multiply_temp_variable = { VIE_naval_cost = 0.33 }
	if = {
		limit = { check_variable = { VIE_sigma_config = 2 } }
		multiply_temp_variable = { VIE_naval_cost = 1.15 }
	}
	else_if = {
		limit = { check_variable = { VIE_sigma_config = 3 } }
		multiply_temp_variable = { VIE_naval_cost = 1.3 }
	}
	if = {
		limit = { has_country_flag = VIE_sigma_late }
		multiply_temp_variable = { VIE_naval_cost = 1.25 }
	}
}

# Funding Gate. Nguoi goi dat truoc VIE_sigma_qty_ordered va VIE_sigma_config.
# Qua cong: tru ngan kho va ky. Truot: event .43.
VIE_naval_sigma_funding_check = {
	VIE_naval_sigma_cost = yes
	if = {
		limit = { VIE_naval_can_fund = yes }
		VIE_naval_contract_sigma = yes
	}
	else = {
		country_event = { id = vie_naval.43 }
	}
}

# Ky hop dong. Thanh toan bang ngan kho; neu VIE_sigma_use_debt thi vay (no = cost x 1,1).
VIE_naval_contract_sigma = {
	set_country_flag = VIE_sigma_contracted
	VIE_naval_sigma_cost = yes
	if = {
		limit = { has_country_flag = VIE_sigma_use_debt }
		set_temp_variable = { debt_change = VIE_naval_cost }
		multiply_temp_variable = { debt_change = 1.1 }
		modify_debt_effect = yes
	}
	else = {
		set_temp_variable = { treasury_change = VIE_naval_cost }
		multiply_temp_variable = { treasury_change = -1 }
		modify_treasury_effect = yes
	}
	# Tau dau giao sau lead: 900 / 1080 / 1260 ngay (x1,0 / 1,2 / 1,4 thoi gian dong).
	# Ky muon (Decision sigma_late): thoi gian giao x 1,25 -> 1125 / 1350 / 1575 ngay.
	if = {
		limit = { has_country_flag = VIE_sigma_late }
		if = {
			limit = { check_variable = { VIE_sigma_config = 2 } }
			country_event = { id = vie_naval.45 days = 1350 }
		}
		else_if = {
			limit = { check_variable = { VIE_sigma_config = 3 } }
			country_event = { id = vie_naval.45 days = 1575 }
		}
		else = {
			country_event = { id = vie_naval.45 days = 1125 }
		}
	}
	else_if = {
		limit = { check_variable = { VIE_sigma_config = 2 } }
		country_event = { id = vie_naval.45 days = 1080 }
	}
	else_if = {
		limit = { check_variable = { VIE_sigma_config = 3 } }
		country_event = { id = vie_naval.45 days = 1260 }
	}
	else = {
		country_event = { id = vie_naval.45 days = 900 }
	}
}

# Giao tau Sigma (goi tu event an .45). Ten tau placeholder. TODO(names).
VIE_naval_deliver_sigma_ship = {
	VIE_naval_ensure_sigma_variant = yes
	if = {
		limit = { check_variable = { VIE_sigma_qty_delivered = 0 } }
		create_ship = { type = frigate_hull_3 equipment_variant = "Sigma Class" creator = VIE name = "Sigma 1" }
	}
	else_if = {
		limit = { check_variable = { VIE_sigma_qty_delivered = 1 } }
		create_ship = { type = frigate_hull_3 equipment_variant = "Sigma Class" creator = VIE name = "Sigma 2" }
	}
	else_if = {
		limit = { check_variable = { VIE_sigma_qty_delivered = 2 } }
		create_ship = { type = frigate_hull_3 equipment_variant = "Sigma Class" creator = VIE name = "Sigma 3" }
	}
	else_if = {
		limit = { check_variable = { VIE_sigma_qty_delivered = 3 } }
		create_ship = { type = frigate_hull_3 equipment_variant = "Sigma Class" creator = VIE name = "Sigma 4" }
	}
	else_if = {
		limit = { check_variable = { VIE_sigma_qty_delivered = 4 } }
		create_ship = { type = frigate_hull_3 equipment_variant = "Sigma Class" creator = VIE name = "Sigma 5" }
	}
	else = {
		create_ship = { type = frigate_hull_3 equipment_variant = "Sigma Class" creator = VIE name = "Sigma 6" }
	}
	add_to_variable = { VIE_sigma_qty_delivered = 1 }
	add_to_variable = { VIE_var_hulls_delivered = 1 }
	# Du so tau: kinh nghiem tich hop theo cau hinh (Domestic +8, Hybrid +4).
	if = {
		limit = { NOT = { check_variable = { VIE_sigma_qty_delivered < VIE_sigma_qty_ordered } } }
		if = {
			limit = { check_variable = { VIE_sigma_config = 3 } }
			add_to_variable = { VIE_var_integration_exp = 8 }
		}
		else_if = {
			limit = { check_variable = { VIE_sigma_config = 2 } }
			add_to_variable = { VIE_var_integration_exp = 4 }
		}
	}
}
