		VIE_init_political_system = yes

		country_event = { id = vie_pol_runtime.1 days = 1 }
	}
}

VIE_politics_forecast = {
	if = {
		limit = { NOT = { has_country_flag = VIE_programme_active } }
		set_variable = { VIE_programme_start_cost = 40 }
		if = { limit = { NOT = { check_variable = { VIE_tier_cadre < 3 } } } set_variable = { VIE_programme_start_cost = 25 } }
		else_if = { limit = { NOT = { check_variable = { VIE_tier_cadre < 2 } } } set_variable = { VIE_programme_start_cost = 30 } }
		else_if = { limit = { NOT = { check_variable = { VIE_tier_cadre < 1 } } } set_variable = { VIE_programme_start_cost = 35 } }
		if = { limit = { NOT = { check_variable = { VIE_leadership_cohesion < 0 } } check_variable = { VIE_leadership_cohesion < 20 } }
			if = { limit = { check_variable = { VIE_programme_start_cost = 25 } } set_variable = { VIE_programme_start_cost = 32 } }
			else_if = { limit = { check_variable = { VIE_programme_start_cost = 30 } } set_variable = { VIE_programme_start_cost = 38 } }
			else_if = { limit = { check_variable = { VIE_programme_start_cost = 35 } } set_variable = { VIE_programme_start_cost = 44 } }
			else_if = { limit = { check_variable = { VIE_programme_start_cost = 40 } } set_variable = { VIE_programme_start_cost = 50 } }
		}
		else_if = { limit = { NOT = { check_variable = { VIE_leadership_cohesion < 20 } } check_variable = { VIE_leadership_cohesion < 40 } }
			if = { limit = { check_variable = { VIE_programme_start_cost = 25 } } set_variable = { VIE_programme_start_cost = 29 } }
			else_if = { limit = { check_variable = { VIE_programme_start_cost = 30 } } set_variable = { VIE_programme_start_cost = 35 } }
			else_if = { limit = { check_variable = { VIE_programme_start_cost = 35 } } set_variable = { VIE_programme_start_cost = 41 } }
			else_if = { limit = { check_variable = { VIE_programme_start_cost = 40 } } set_variable = { VIE_programme_start_cost = 46 } }
		}
		else_if = { limit = { NOT = { check_variable = { VIE_leadership_cohesion < 40 } } check_variable = { VIE_leadership_cohesion < 55 } }
			if = { limit = { check_variable = { VIE_programme_start_cost = 25 } } set_variable = { VIE_programme_start_cost = 27 } }
			else_if = { limit = { check_variable = { VIE_programme_start_cost = 30 } } set_variable = { VIE_programme_start_cost = 32 } }
			else_if = { limit = { check_variable = { VIE_programme_start_cost = 35 } } set_variable = { VIE_programme_start_cost = 37 } }
			else_if = { limit = { check_variable = { VIE_programme_start_cost = 40 } } set_variable = { VIE_programme_start_cost = 42 } }
		}
		else_if = { limit = { NOT = { check_variable = { VIE_leadership_cohesion < 55 } } check_variable = { VIE_leadership_cohesion < 70 } }
			if = { limit = { check_variable = { VIE_programme_start_cost = 25 } } set_variable = { VIE_programme_start_cost = 25 } }
			else_if = { limit = { check_variable = { VIE_programme_start_cost = 30 } } set_variable = { VIE_programme_start_cost = 30 } }
			else_if = { limit = { check_variable = { VIE_programme_start_cost = 35 } } set_variable = { VIE_programme_start_cost = 35 } }
			else_if = { limit = { check_variable = { VIE_programme_start_cost = 40 } } set_variable = { VIE_programme_start_cost = 40 } }
		}
		else_if = { limit = { NOT = { check_variable = { VIE_leadership_cohesion < 70 } } check_variable = { VIE_leadership_cohesion < 101 } }
			if = { limit = { check_variable = { VIE_programme_start_cost = 25 } } set_variable = { VIE_programme_start_cost = 23 } }
			else_if = { limit = { check_variable = { VIE_programme_start_cost = 30 } } set_variable = { VIE_programme_start_cost = 27 } }
			else_if = { limit = { check_variable = { VIE_programme_start_cost = 35 } } set_variable = { VIE_programme_start_cost = 32 } }
			else_if = { limit = { check_variable = { VIE_programme_start_cost = 40 } } set_variable = { VIE_programme_start_cost = 36 } }
		}
		set_variable = { VIE_programme_prepare_days = 30 }
		if = { limit = { VIE_is_central_zone = yes } set_variable = { VIE_programme_prepare_days = 15 } }
		if = { limit = { check_variable = { VIE_leadership_cohesion < 20 } } add_to_variable = { VIE_programme_prepare_days = 60 } }
		set_variable = { VIE_programme_forecast_standard = VIE_programme_prepare_days }
		set_variable = { VIE_programme_forecast_fast = VIE_programme_prepare_days }
		if = { limit = { has_country_flag = VIE_programme_inspection }
			add_to_variable = { VIE_programme_forecast_standard = 720 }
			add_to_variable = { VIE_programme_forecast_fast = 630 }
		} else = {
			add_to_variable = { VIE_programme_forecast_standard = 630 }
			add_to_variable = { VIE_programme_forecast_fast = 540 }
	}}

}

VIE_politics_e104_tick = {
	if = { limit = { has_country_flag = VIE_e104_signing }
		subtract_from_variable = { VIE_e104_sign_remaining = 1 }
		if = { limit = { check_variable = { VIE_e104_sign_remaining < 1 } }
			clr_country_flag = VIE_e104_signing
			VIE_politics_e104_open = yes
		}
	} else_if = { limit = { has_country_flag = VIE_e104_signed NOT = { has_country_flag = VIE_e104_terminated } }
		add_to_variable = { VIE_e104_contract_elapsed = 1 }
		if = { limit = { NOT = { check_variable = { VIE_e104_contract_elapsed < 90 } } NOT = { has_country_flag = VIE_e104_check1_done } }
			if = { limit = { has_country_flag = VIE_e104_check1_paid has_completed_focus = VIE_publish_procurement_audits }
				set_country_flag = VIE_e104_check1_done
			} else = { VIE_politics_e104_fail = yes }
		}
		if = { limit = { has_country_flag = VIE_e104_signed NOT = { check_variable = { VIE_e104_contract_elapsed < 180 } } NOT = { has_country_flag = VIE_e104_check2_done } }
			if = { limit = { has_country_flag = VIE_e104_check2_paid has_country_flag = VIE_e104_check1_done }
				set_country_flag = VIE_e104_check2_done
				set_temp_variable = { treasury_change = VIE_e104_remaining_support }
				modify_treasury_effect = yes
				set_variable = { VIE_e104_remaining_support = 0 }
			} else = { VIE_politics_e104_fail = yes }
		}
		if = { limit = { has_country_flag = VIE_e104_training_running }
			add_to_variable = { VIE_e104_training_elapsed = 1 }
			if = { limit = { NOT = { check_variable = { VIE_e104_training_elapsed < 90 } } }
				clr_country_flag = VIE_e104_training_running
				set_country_flag = VIE_e104_training_complete
			}
		}
		if = { limit = { has_country_flag = VIE_e104_signed has_country_flag = VIE_e104_check2_done has_country_flag = VIE_e104_training_complete NOT = { has_country_flag = VIE_e104_training_rewarded } }
			set_country_flag = VIE_e104_training_rewarded
			clr_country_flag = VIE_e104_rights_unused
			add_political_power = 25
			clr_country_flag = VIE_e104_signed
		}
	}
}

VIE_politics_refresh_targets = {
	if = {
		limit = { has_country_flag = VIE_programme_active }
		set_variable = { VIE_programme_targets_met = 0 }
		if = {
			limit = { has_country_flag = VIE_programme_cadre }
			if = {
				limit = { has_completed_focus = VIE_publish_responsibility_rules }
				add_to_variable = { VIE_programme_targets_met = 1 }}
			if = {
				limit = { check_variable = { VIE_programme_step > 1 } }
				add_to_variable = { VIE_programme_targets_met = 1 }}
			if = {
				limit = { check_variable = { VIE_programme_step > 2 } }
				add_to_variable = { VIE_programme_targets_met = 1 }}
			if = {
				limit = { has_country_flag = VIE_programme_review_done }
				add_to_variable = { VIE_programme_targets_met = 1 }}
		}
		if = {
			limit = { has_country_flag = VIE_programme_admin }
			if = {
				limit = { has_completed_focus = VIE_pilot_one_stop_service }
				add_to_variable = { VIE_programme_targets_met = 1 }}
			if = {
				limit = { check_variable = { VIE_programme_step > 1 } }
				add_to_variable = { VIE_programme_targets_met = 1 }}
			if = {
				limit = { has_country_flag = VIE_programme_review_done }
				add_to_variable = { VIE_programme_targets_met = 1 }}
			if = {
				limit = { has_country_flag = VIE_programme_review_done NOT = { check_variable = { treasury_rate < 0 } } }
				add_to_variable = { VIE_programme_targets_met = 1 }}
		}
		if = {
			limit = { has_country_flag = VIE_programme_inspection }
			if = {
				limit = { check_variable = { VIE_programme_step > 0 } }
				add_to_variable = { VIE_programme_targets_met = 1 }}
			if = {
				limit = { check_variable = { VIE_programme_step > 2 } }
				add_to_variable = { VIE_programme_targets_met = 1 }}
			if = {
				limit = { has_country_flag = VIE_programme_review_done }
				add_to_variable = { VIE_programme_targets_met = 1 }}
			if = {
				limit = { has_country_flag = VIE_programme_review_done VIE_politics_corruption_verified = yes }
				add_to_variable = { VIE_programme_targets_met = 1 }}
		}
		if = {
			limit = { has_country_flag = VIE_programme_social }
			if = {
				limit = { has_completed_focus = VIE_establish_grievance_hearings }
				add_to_variable = { VIE_programme_targets_met = 1 }}
			if = {
				limit = { check_variable = { VIE_programme_step > 1 } }
				add_to_variable = { VIE_programme_targets_met = 1 }}
			if = {
				limit = { has_country_flag = VIE_programme_review_done }
				add_to_variable = { VIE_programme_targets_met = 1 }}
			if = {
				limit = { has_country_flag = VIE_programme_review_done check_variable = { VIE_programme_gdp_target > 0 } NOT = { check_variable = { var = gdp_per_capita value = VIE_programme_gdp_target compare = less_than } } }
				add_to_variable = { VIE_programme_targets_met = 1 }}
		}
	}

}

VIE_politics_job_finish = {
	if = { limit = { has_country_flag = VIE_programme_active has_country_flag = VIE_programme_job_running check_variable = { var = VIE_programme_job_epoch value = VIE_programme_epoch compare = equals } }
		clr_country_flag = VIE_programme_job_running
		add_to_variable = { VIE_programme_step = 1 }
		add_to_variable = { VIE_programme_targets_met = 1 }
		if = { limit = { has_country_flag = VIE_programme_inspection check_variable = { VIE_programme_step = 3 } NOT = { has_idea = corruption_level_01 } }
			decrease_corruption = yes
			VIE_politics_read_corruption = yes
		}
		if = { limit = { has_country_flag = VIE_programme_job_review }
			set_country_flag = VIE_programme_review_done
			if = { limit = { OR = { has_country_flag = VIE_programme_admin has_country_flag = VIE_programme_social } } add_to_variable = { VIE_programme_targets_met = 1 } }
		}
	}
}

VIE_politics_evaluate = {
	if = {
		limit = { has_country_flag = VIE_programme_active }

VIE_politics_programme_tick = {
	if = { limit = { has_country_flag = VIE_programme_active }
		VIE_politics_read_corruption = yes
		add_to_variable = { VIE_programme_elapsed = 1 }
		if = { limit = { has_country_flag = VIE_programme_job_delay check_variable = { var = VIE_programme_elapsed value = VIE_programme_prepare_days compare = greater_than } }
			subtract_from_variable = { VIE_programme_delay_remaining = 1 }
			if = { limit = { check_variable = { VIE_programme_delay_remaining < 1 } } clr_country_flag = VIE_programme_job_delay }
		}
		if = { limit = { has_country_flag = VIE_programme_job_running NOT = { has_country_flag = VIE_programme_job_delay } VIE_politics_job_can_progress = yes }
			if = { limit = { has_country_flag = VIE_programme_job_review NOT = { VIE_politics_review_condition = yes } }
				set_variable = { VIE_programme_job_remaining = 90 }
			} else = { subtract_from_variable = { VIE_programme_job_remaining = 1 } }
			if = { limit = { check_variable = { VIE_programme_job_remaining < 1 } } VIE_politics_job_finish = yes }
		}
		VIE_politics_read_corruption = yes
		VIE_politics_refresh_targets = yes
		VIE_politics_remaining_forecast = yes
		VIE_politics_offer_e102 = yes
		if = { limit = { has_country_flag = VIE_programme_inspection NOT = { check_variable = { VIE_programme_elapsed < 450 } } NOT = { has_country_flag = VIE_e103_progress_notified } }
			set_country_flag = VIE_e103_progress_notified
			country_event = { id = vie_pol.103 }
		}
		if = { limit = { has_country_flag = VIE_programme_inspection has_country_flag = VIE_e102_data_access_unlocked NOT = { check_variable = { VIE_programme_elapsed < 500 } } NOT = { has_country_flag = VIE_e104_offer_shown } NOT = { has_country_flag = VIE_e104_signed } }
			set_country_flag = VIE_e104_offer_shown
			set_variable = { VIE_e104_offer_epoch = VIE_programme_epoch }
			VIE_politics_dispatch_104 = yes
		}
		VIE_politics_evaluate = yes
	}
}
