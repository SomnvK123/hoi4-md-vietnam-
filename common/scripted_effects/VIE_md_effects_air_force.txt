# ---- Slot chuong trinh luc luong (toi da 2 chay cung luc) ----
VIE_airf_program_start = {
	add_to_variable = { VIE_var_airf_program_active = 1 }
	clamp_variable = { var = VIE_var_airf_program_active min = 0 max = 2 }
}
VIE_airf_program_end = {
	add_to_variable = { VIE_var_airf_program_active = -1 }
	clamp_variable = { var = VIE_var_airf_program_active min = 0 max = 2 }
}
# Tu chua hang thang: khong con timed idea chuong trinh nao va khong co co cho (Decision vua bam, chua chon option) -> ve 0.
VIE_airf_slot_heal = {
	if = {
		limit = {
			NOT = { has_idea = VIE_airf_prog_fighters }
			NOT = { has_idea = VIE_airf_prog_sam }
			NOT = { has_idea = VIE_airf_prog_coord }
			NOT = { has_idea = VIE_airf_prog_first }
			NOT = { has_idea = VIE_airf_prog_capstone }
			NOT = { has_country_flag = VIE_airf_d1_pending }
			NOT = { has_country_flag = VIE_airf_d2_pending }
			NOT = { has_country_flag = VIE_airf_d4_pending }
			check_variable = { VIE_var_airf_program_active > 0 }
		}
		set_variable = { VIE_var_airf_program_active = 0 }
	}
}

# ---- Tien (ty USD) va thoi gian (thang -> ngay) cua chuong trinh, dung bien tam trong cung mot option ----
VIE_airf_cost_set = {
	set_temp_variable = { VIE_airf_cost = $C$ }
}
VIE_airf_charge = {
	set_temp_variable = { treasury_change = VIE_airf_cost }
	multiply_temp_variable = { treasury_change = -1 }
	modify_treasury_effect = yes
}
VIE_airf_time_set = {
	set_temp_variable = { VIE_airf_months = $M$ }
}
# Tham so IDEA (VIE_airf_prog_*), EV (so event an vie_air_force.<EV>). Hen timed idea va event hoan tat cung mot moc.
VIE_airf_start_timer = {
	set_temp_variable = { VIE_airf_end_days = VIE_airf_months }
	multiply_temp_variable = { VIE_airf_end_days = 30 }
	round_temp_variable = VIE_airf_end_days
	add_timed_idea = { idea = $IDEA$ days = VIE_airf_end_days }
	country_event = { id = vie_air_force.$EV$ days = VIE_airf_end_days }
}

# Mo nhanh hoc thuyet (A1 / B1 / C1) so voi co cau luc luong da chon o D-D (VIE_airf_force_priority: 1 Phong thu lanh tho,
# 2 Can bang, 3 Tam xa; 0 = chua chon). Khop -> MIS +1%; lech -> timed idea 365 ngay; Can bang/chua chon -> trung tinh.
# A hop co cau 1, lech co cau 3. B va C hop co cau 3, lech co cau 1.

VIE_airf_branch_fit_a = {
	if = {
		limit = { check_variable = { VIE_airf_force_priority = 1 } }
		VIE_airf_add_mis = { V = 0.01 }
	}
	else_if = {
		limit = { check_variable = { VIE_airf_force_priority = 3 } }
		add_timed_idea = { idea = VIE_airf_branch_mismatch_idea days = 365 }
	}
}

# Decision D-A: huan luyen luc luong tiem kich (event vie_air_force.1 .2, ket thuc .61).
# Nguoi goi dat truoc VIE_airf_fighter_level (1 co ban / 2 chuyen sau) va VIE_airf_fighter_specialty (1 khong chien / 2 tan cong dat-bien).
# Chi phi 0,40 / 0,60 ty, thoi luong 12 / 18 thang.
########################################################################
VIE_airf_d1_start = {
	VIE_airf_cost_set = { C = 0.4 }
	VIE_airf_time_set = { M = 12 }
	if = {
		limit = { check_variable = { VIE_airf_fighter_level = 2 } }
		VIE_airf_cost_set = { C = 0.6 }
		VIE_airf_time_set = { M = 18 }
	}
	VIE_airf_charge = yes
	clr_country_flag = VIE_airf_d1_pending
	VIE_airf_start_timer = { IDEA = VIE_airf_prog_fighters EV = 61 }
}

# Ket thuc D-A. Muc 2 = 1,5 lan muc 1.
VIE_airf_d1_finish = {
	if = {
		limit = { check_variable = { VIE_airf_fighter_specialty = 2 } }
		if = {
			limit = { check_variable = { VIE_airf_fighter_level = 2 } }
			VIE_airf_add_cas = { V = 0.045 }
			VIE_airf_add_atk = { V = 0.015 }
		}
		else = {
			VIE_airf_add_cas = { V = 0.03 }
			VIE_airf_add_atk = { V = 0.01 }
		}
	}
	else = {
		if = {
			limit = { check_variable = { VIE_airf_fighter_level = 2 } }
			VIE_airf_add_sup = { V = 0.045 }
		}
		else = {
			VIE_airf_add_sup = { V = 0.03 }
		}
	}
	set_country_flag = VIE_airf_d1_done
	VIE_airf_program_end = yes
}
