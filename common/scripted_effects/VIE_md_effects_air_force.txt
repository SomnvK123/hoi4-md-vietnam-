VIE_airf_slot_heal = {
	if = {
		limit = {
			NOT = { has_idea = VIE_airf_prog_fighters }
			NOT = { has_idea = VIE_airf_prog_sam }
			NOT = { has_idea = VIE_airf_prog_coord }
			NOT = { has_idea = VIE_airf_prog_first }
			NOT = { has_idea = VIE_airf_prog_capstone }
			NOT = { has_country_flag = VIE_airf_d1_pending }

		set_variable = { VIE_var_airf_program_active = 0 }
	}
}

VIE_event_scheduler_airf = {
	if = {
		limit = { NOT = { check_variable = { VIE_catch_up = 1 } } }
		if = {
			limit = {
				date > 2011.5.31
				NOT = { has_country_flag = VIE_airf_ms_su30_923 }
			}
			if = {
				limit = { NOT = { has_country_flag = VIE_popup_cd } }
				set_country_flag = VIE_airf_ms_su30_923
				set_country_flag = { flag = VIE_popup_cd value = 1 days = 45 }
				country_event = { id = vie_air_force.70 days = 3 random_days = 15 }
			}
			else_if = {
				limit = { date > 2011.11.30 }
				set_country_flag = VIE_airf_ms_su30_923
				VIE_airf_ms1_apply = yes
			}
		}
		if = {
			limit = {
				date > 2016.2.29

			}
			if = {
				limit = { NOT = { has_country_flag = VIE_popup_cd } }
