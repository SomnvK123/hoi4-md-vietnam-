# chi dat co moc cho nhung moc da qua, khong ban event, khong ap hieu ung.
########################################################################
VIE_event_scheduler_p17 = {
	if = {
		limit = { NOT = { check_variable = { VIE_catch_up = 1 } } }
		if = {
			limit = {
				date > 2022.1.16
				NOT = { has_country_flag = VIE_lf_ms_nq05 }
			}
			if = {
				limit = { NOT = { has_country_flag = VIE_popup_cd } }
				hidden_effect = { set_country_flag = VIE_lf_ms_nq05 }
				hidden_effect = { set_country_flag = { flag = VIE_popup_cd value = 1 days = 45 } }
				country_event = { id = vie_lf.1 days = 3 random_days = 15 }
			}
			else_if = {
				limit = { date > 2022.7.31 }
				hidden_effect = { set_country_flag = VIE_lf_ms_nq05 }
				VIE_lf_ms1_apply = yes
			}
		}
		if = {
			limit = {
				date > 2022.12.19
				NOT = { has_country_flag = VIE_lf_ms_nq1657 }
			}
			hidden_effect = { set_country_flag = VIE_lf_ms_nq1657 }
			country_event = { id = vie_lf.2 days = 3 random_days = 15 }
		}
		if = {
			limit = {
				date > 2023.12.1
				NOT = { has_country_flag = VIE_lf_ms_corps12 }
			}
			if = {
				limit = { NOT = { has_country_flag = VIE_popup_cd } }
				hidden_effect = { set_country_flag = VIE_lf_ms_corps12 }
				hidden_effect = { set_country_flag = { flag = VIE_popup_cd value = 1 days = 45 } }
				country_event = { id = vie_lf.3 days = 3 random_days = 15 }
			}
			else_if = {
				limit = { date > 2024.5.31 }
				hidden_effect = { set_country_flag = VIE_lf_ms_corps12 }
				VIE_lf_ms3_apply = yes
			}
		}
		if = {
			limit = {
				date > 2024.12.14
				NOT = { has_country_flag = VIE_lf_ms_corps34 }
			}
			hidden_effect = { set_country_flag = VIE_lf_ms_corps34 }
			country_event = { id = vie_lf.4 days = 3 random_days = 15 }
		}
		if = {
			limit = {
				date > 2025.2.4
				NOT = { has_country_flag = VIE_lf_ms_logistics }
			}
			if = {
				limit = { NOT = { has_country_flag = VIE_popup_cd } }
				hidden_effect = { set_country_flag = VIE_lf_ms_logistics }
				hidden_effect = { set_country_flag = { flag = VIE_popup_cd value = 1 days = 45 } }
				country_event = { id = vie_lf.5 days = 3 random_days = 15 }
			}
			else_if = {
				limit = { date > 2025.8.31 }
				hidden_effect = { set_country_flag = VIE_lf_ms_logistics }
				VIE_lf_ms5_apply = yes
			}
		}
	}
	else = {
		if = {
			limit = { date > 2022.1.16 }
			hidden_effect = { set_country_flag = VIE_lf_ms_nq05 }
		}
		if = {
			limit = { date > 2022.12.19 }
			hidden_effect = { set_country_flag = VIE_lf_ms_nq1657 }
		}
		if = {
			limit = { date > 2023.12.1 }
			hidden_effect = { set_country_flag = VIE_lf_ms_corps12 }
		}
		if = {
			limit = { date > 2024.12.14 }
			hidden_effect = { set_country_flag = VIE_lf_ms_corps34 }
		}
		if = {
			limit = { date > 2025.2.4 }
			hidden_effect = { set_country_flag = VIE_lf_ms_logistics }
		}
	}
}
