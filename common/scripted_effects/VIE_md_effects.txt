# South China Sea tension (0-3). Caller: set_temp_variable = { VIE_tension_add = N }
########################################################################
VIE_add_scs_tension = {
	hidden_effect = {
		add_to_variable = { VIE_scs_tension = VIE_tension_add }
		clamp_variable = { var = VIE_scs_tension min = 0 max = 3 }
	}
	if = {
		limit = { check_variable = { VIE_scs_tension > 1 } }
		hidden_effect = { set_country_flag = VIE_scs_escalated }
	}
}

#   VIE_create_leader_<x>  only creates the character (used by set_leader_VIE to rebuild the

#   VIE_new_leader_<x>     Congress hand-over: kill, create, remember who holds the post (VIE_gs)

# Rebuild whoever holds the post (called from set_leader_VIE after MD's set_leader killed the leader)
VIE_recreate_general_secretary = {
	if = { limit = { check_variable = { VIE_gs = 1 } } VIE_create_leader_phieu = yes }
	else_if = { limit = { check_variable = { VIE_gs = 2 } } VIE_create_leader_manh = yes }
	else_if = { limit = { check_variable = { VIE_gs = 3 } } VIE_create_leader_trong = yes }
	else_if = { limit = { check_variable = { VIE_gs = 4 } } VIE_create_leader_dung = yes }
	else_if = { limit = { check_variable = { VIE_gs = 5 } } VIE_create_leader_to_lam = yes }
	else = { VIE_create_leader_restored_party = yes }
}

# Party popularity helper (MD change_relative_party_popularity). Adds AMOUNT to the outlook AND the party,
# so it also works for parties whose outlook has ~0 popularity.  Usage:
#   VIE_boost_party = { PARTY = 20 AMOUNT = 0.15 }
########################################################################
########################################################################
# Single gateway for regime changes.
# Caller: set_temp_variable = { rul_party_temp = <MD party index> }  VIE_transition_regime = yes
# Party indexes: 19 CPV(Neutral_Communism) 4 Communist-State 7 Autocracy (20 Nat_Populism and 21 Nat_Fascism are not used by the submod)
#                22 Nat_Autocracy 16 Neutral_Libertarian 15 oligarchism 13 Neutral_Autocracy
#                0 Western_Autocracy 1 conservatism 2 liberalism 3 socialism
# Idempotent: does nothing if the country already has that ruling party.
########################################################################
VIE_transition_regime = {
	if = {
		limit = { NOT = { check_variable = { ruling_party = rul_party_temp } } }

		# The new ruling party must not be banned (VIE starts with all parties but the ruling one banned)
		set_temp_variable = { party_index = rul_party_temp }
		unban_party_scripted_call = yes

		# Elections only for the democratic parties (change_ruling_party_effect reads disable_elections)
		if = {
			limit = {
				OR = {
					check_variable = { rul_party_temp = 1 }
					check_variable = { rul_party_temp = 2 }
					check_variable = { rul_party_temp = 3 }
					check_variable = { rul_party_temp = 14 }
					check_variable = { rul_party_temp = 18 }
				}
			}
			set_temp_variable = { disable_elections = 0 }
		}
		else = { set_temp_variable = { disable_elections = 1 } }

		# Drop every VIE balance of power before the ruling party changes
		if = {
			limit = { has_country_flag = VIE_bop_active }
			remove_power_balance = { id = VIE_party_balance }
			hidden_effect = { clr_country_flag = VIE_bop_active }
		}

		# Leaving the hardline party (4): drop its ideas and the reform pressure while ruling_party is still 4
		if = {
			limit = { check_variable = { ruling_party = 4 } }
			VIE_hl_cleanup = yes
		}

		set_temp_variable = { change_leader_temp = 1 }
		change_ruling_party_effect = yes   # MD: sets ruling_party, set_politics, runs set_leader_VIE

		# Balance of power of the new regime
		if = {
			limit = {
				OR = {
					check_variable = { ruling_party = 19 }
					check_variable = { ruling_party = 4 }
				}
			}
			hidden_effect = { clr_country_flag = VIE_party_rule_ended }
			set_power_balance = {
				id = VIE_party_balance
				left_side = VIE_conservative_side
				right_side = VIE_reform_side
				set_value = 0
			}
			hidden_effect = { set_country_flag = VIE_bop_active }
		}
		else = {
			hidden_effect = { set_country_flag = VIE_party_rule_ended }
			# Party rule is over: set_leader_VIE must build the leader of a later return to slot 19 itself
			hidden_effect = { clr_country_flag = VIE_congress_controls_leader }
			hidden_effect = { set_variable = { VIE_gs = 0 } }
			# Four Nos do not survive these regimes
			if = {
				limit = {
					OR = {
						check_variable = { ruling_party = 22 }
						check_variable = { ruling_party = 0 }
					}
				}
				if = { limit = { has_idea = VIE_four_nos } remove_ideas = VIE_four_nos }
				if = { limit = { has_idea = VIE_non_alignment_policy } remove_ideas = VIE_non_alignment_policy }
			}
		}

		# Blocks other doors for 6 months so two regime changes cannot stack
		hidden_effect = { set_country_flag = { flag = VIE_regime_changed_recently value = 1 days = 180 } }
		hidden_effect = { set_variable = { VIE_regime = ruling_party } }
	}
}

# Pop-up scheduler. Called monthly (see VIE_md_on_actions.txt).
#  Class A (turning point): always fires, sets the 45 day cooldown.
#  Class B (choice event):  fires only when the cooldown is free; if it is still blocked
#                           after DATE_TO the historical result is applied silently (FALLBACK).
# Catch-up mode (temp variable VIE_catch_up = 1): only sets the "already happened" flags, fires nothing.
# Used when a civil-war winner takes over the VIE lineage (see VIE_catch_up_schedule).
########################################################################
VIE_event_scheduler = {
	# One-time init (hidden event vie_pol.1)
	if = {
		limit = { NOT = { has_country_flag = VIE_md_init_done } }
		hidden_effect = { set_country_flag = VIE_md_init_done }
		if = {
			limit = { NOT = { check_variable = { VIE_catch_up = 1 } } }
			country_event = { id = vie_pol.1 }
		}
	}

	# Party Congresses: only while the Party rules
	if = {
		limit = { VIE_party_rule_active = yes }
		if = {
			limit = {
				date > 2001.3.31
				NOT = { has_country_flag = VIE_sched_congress_9 }
			}
			hidden_effect = { set_country_flag = VIE_sched_congress_9 }
			VIE_start_congress_term_9 = yes
			if = {
				limit = { NOT = { check_variable = { VIE_catch_up = 1 } } }
				hidden_effect = { set_country_flag = { flag = VIE_popup_cd value = 1 days = 45 } }
				country_event = { id = vie_pol.2 days = 3 random_days = 15 }
			}
		}
		if = {
			limit = {
				date > 2006.3.31
				NOT = { has_country_flag = VIE_sched_congress_10 }
			}
			hidden_effect = { set_country_flag = VIE_sched_congress_10 }
			VIE_start_congress_term_10 = yes
			if = {
				limit = { NOT = { check_variable = { VIE_catch_up = 1 } } }
				hidden_effect = { set_country_flag = { flag = VIE_popup_cd value = 1 days = 45 } }
				country_event = { id = vie_pol.3 days = 3 random_days = 15 }
			}
		}
		if = {
			limit = {
				date > 2010.12.31
				NOT = { has_country_flag = VIE_sched_congress_11 }
			}
			hidden_effect = { set_country_flag = VIE_sched_congress_11 }
			VIE_start_congress_term_11 = yes
			if = {
				limit = { NOT = { check_variable = { VIE_catch_up = 1 } } }
				hidden_effect = { set_country_flag = { flag = VIE_popup_cd value = 1 days = 45 } }
				country_event = { id = vie_pol.4 days = 3 random_days = 15 }
			}
		}
		if = {
			limit = {
				date > 2015.12.31
				NOT = { has_country_flag = VIE_sched_congress_12 }
			}
			hidden_effect = { set_country_flag = VIE_sched_congress_12 }
			VIE_start_congress_term_12 = yes
			if = {
				limit = { NOT = { check_variable = { VIE_catch_up = 1 } } }
				hidden_effect = { set_country_flag = { flag = VIE_popup_cd value = 1 days = 45 } }
				country_event = { id = vie_pol.5 days = 3 random_days = 15 }
			}
		}
	}

	# 2019 Defence White Paper: informal non-alignment becomes "Four Nos" (class C: no pop-up)
	if = {
		limit = {
			date > 2019.10.31
			NOT = { has_country_flag = VIE_sched_four_nos }
		}
		hidden_effect = { set_country_flag = VIE_sched_four_nos }
		if = {
			limit = { has_idea = VIE_non_alignment_policy }
			swap_ideas = { remove_idea = VIE_non_alignment_policy add_idea = VIE_four_nos }
		}
	}

	# Class B: Nam Cam case (2002-03)
	if = {
		limit = {
			date > 2002.8.31
			NOT = { has_country_flag = VIE_sched_nam_cam }
		}
		if = {
			limit = { check_variable = { VIE_catch_up = 1 } }
			hidden_effect = { set_country_flag = VIE_sched_nam_cam }
		}
		else_if = {
			limit = { NOT = { has_country_flag = VIE_popup_cd } }
			hidden_effect = { set_country_flag = VIE_sched_nam_cam }
			hidden_effect = { set_country_flag = { flag = VIE_popup_cd value = 1 days = 45 } }
			country_event = { id = vie_eco.2 days = 3 random_days = 20 }
		}
		else_if = {
			limit = { date > 2003.2.1 }
			hidden_effect = { set_country_flag = VIE_sched_nam_cam }
			VIE_fb_nam_cam = yes
		}
	}

	# Class A: HD-981 oil rig standoff (May 2014). Fires regardless of regime.
	if = {
		limit = {
			date > 2014.4.30
			NOT = { has_country_flag = VIE_sched_hd981 }
		}
		hidden_effect = { set_country_flag = VIE_sched_hd981 }
		if = {
			limit = { NOT = { check_variable = { VIE_catch_up = 1 } } }
			hidden_effect = { set_country_flag = { flag = VIE_popup_cd value = 1 days = 45 } }
			country_event = { id = vie_scs.1 days = 3 random_days = 15 }
		}
	}
}
