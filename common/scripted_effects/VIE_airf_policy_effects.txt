VIE_airf_research_1_reward = {
	if = {
		limit = { NOT = { has_country_flag = VIE_airf_research_1_granted } }
		if = {
			limit = { NOT = { AND = { has_country_flag = VIE_airf_policy_selected check_variable = { VIE_airf_policy_priority = 1 } } } }
			add_tech_bonus = { name = VIE_airf_policy_1_bonus bonus = 0.25 uses = 1 category = CAT_surface_to_air_missiles }
		}
		set_country_flag = VIE_airf_research_1_granted
	}
}
