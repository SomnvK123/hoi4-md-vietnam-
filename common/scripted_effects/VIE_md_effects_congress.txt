# Party Congress spine (see VIE_congress_spine_redesign.md). Parameterized helper effect.

########################################################################
# Called from the six Congress scheduler blocks (VIE_md_effects.txt / _p2.txt), right after
# each sets its own VIE_sched_congress_N flag - including in catch-up mode, so a civil-war
# winner taking over the VIE lineage still lands on the correct term.
# Sets the term variable and drops whichever term-spirit idea the previous Congress granted.
# Checks has_idea before each remove_ideas (mod convention) instead of tracking which one
# (or which variant) was actually active.
# Usage: VIE_start_congress_term = { N = 12 }
########################################################################
VIE_start_congress_term = {
	set_variable = { VIE_congress_term = $N$ }

	if = { limit = { has_idea = VIE_resolution_9_idea } remove_ideas = VIE_resolution_9_idea }
	if = { limit = { has_idea = VIE_resolution_9_cons_idea } remove_ideas = VIE_resolution_9_cons_idea }
	if = { limit = { has_idea = VIE_resolution_10_idea } remove_ideas = VIE_resolution_10_idea }
	if = { limit = { has_idea = VIE_resolution_11_idea } remove_ideas = VIE_resolution_11_idea }
	if = { limit = { has_idea = VIE_resolution_12_idea } remove_ideas = VIE_resolution_12_idea }
	if = { limit = { has_idea = VIE_resolution_12_dev_idea } remove_ideas = VIE_resolution_12_dev_idea }
	if = { limit = { has_idea = VIE_resolution_13_idea } remove_ideas = VIE_resolution_13_idea }
	if = { limit = { has_idea = VIE_resolution_14_idea } remove_ideas = VIE_resolution_14_idea }
}
