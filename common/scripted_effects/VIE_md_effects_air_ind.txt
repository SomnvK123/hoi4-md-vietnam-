# ---- Slot chuong trinh (0-2) -------------------------------------------------------------------------------
VIE_apm_program_start = {
	add_to_variable = { VIE_var_apm_active = 1 }
	clamp_variable = { var = VIE_var_apm_active min = 0 max = 2 }
}
VIE_apm_program_end = {
	add_to_variable = { VIE_var_apm_active = -1 }
	clamp_variable = { var = VIE_var_apm_active min = 0 max = 2 }
}
# Tu chua: neu khong con timed idea chuong trinh nao thi bo dem ve 0 (mat event hoan tat, load save cu...).
# Goi hang thang (buoc 7 noi vao scheduler).
VIE_apm_slot_heal = {
	if = {
		limit = {
			NOT = { has_idea = VIE_apm_prog_a32 }
			NOT = { has_idea = VIE_apm_prog_a31 }
			NOT = { has_idea = VIE_apm_prog_radar }
			NOT = { has_idea = VIE_apm_prog_integ }
			NOT = { has_idea = VIE_apm_prog_uav }
			NOT = { has_country_flag = VIE_apm_a32_pending }
			NOT = { has_country_flag = VIE_apm_a31_pending }
			NOT = { has_country_flag = VIE_apm_radar_pending }
			NOT = { has_country_flag = VIE_apm_integ_pending }
			NOT = { has_country_flag = VIE_apm_uav_pending }
			check_variable = { VIE_var_apm_active > 0 }
		}
		set_variable = { VIE_var_apm_active = 0 }
	}
}

		}

			limit = { check_variable = { VIE_apm_a32_choice = 2 } }

		}
	}
	else_if = {
		limit = { check_variable = { VIE_apm_a32_tier = 1 } }
		VIE_ap_ensure_af_modifier = yes
		custom_effect_tooltip = {
			localization_key = modifies_dynamic_modifier_tt

		}

	else_if = {

			localization_key = modifies_dynamic_modifier_tt

	}

	clamp_variable = { var = VIE_apm_a32_tier min = 0 max = 3 }
	VIE_apm_program_end = yes
