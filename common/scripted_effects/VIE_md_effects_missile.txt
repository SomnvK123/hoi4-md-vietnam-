VIE_msl_naval_xp_5 = {
	if = {
		limit = { has_selected_naval_grand_doctrine = yes }
		add_mastery = { amount = 5 folder = naval }
	}
	else = { navy_experience = 5 }
}

VIE_msl_refresh_spirits = {
	if = {
		limit = { has_country_flag = VIE_msl_m09_reward_applied }
		if = { limit = { has_idea = VIE_msl_sustainment_idea } remove_ideas = VIE_msl_sustainment_idea }
		if = { limit = { has_idea = VIE_msl_industry_idea } remove_ideas = VIE_msl_industry_idea }
		if = { limit = { NOT = { has_idea = VIE_msl_integrated_idea } } add_ideas = VIE_msl_integrated_idea }
	}
	else = {
		if = { limit = { has_country_flag = VIE_msl_m04_reward_applied NOT = { has_idea = VIE_msl_sustainment_idea } } add_ideas = VIE_msl_sustainment_idea }
		if = { limit = { has_country_flag = VIE_msl_m08_reward_applied NOT = { has_idea = VIE_msl_industry_idea } } add_ideas = VIE_msl_industry_idea }
	}
}

VIE_msl_m00_reward = {
	if = {
		limit = { original_tag = VIE NOT = { has_country_flag = VIE_msl_m00_reward_applied } }
		set_country_flag = VIE_msl_m00_reward_applied
		set_temp_variable = { treasury_change = -1 }
		modify_treasury_effect = yes
		add_tech_bonus = { name = VIE_msl_national_program bonus = 0.1 uses = 1 category = CAT_missiles }
		set_country_flag = VIE_msl_program_started
	}
}
