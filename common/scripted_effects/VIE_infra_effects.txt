# Infrastructure events scheduled through the shared VIE popup cooldown.
VIE_event_scheduler_infra = {
	if = {
		limit = {
			date > 2020.3.31
			NOT = { has_country_flag = VIE_sched_vna_covid }
		}
		if = {
			limit = { check_variable = { VIE_catch_up = 1 } }
			set_country_flag = VIE_sched_vna_covid
		}
		else_if = {
			limit = { NOT = { has_country_flag = VIE_popup_cd } }
			set_country_flag = VIE_sched_vna_covid
			set_country_flag = { flag = VIE_popup_cd value = 1 days = 45 }
			country_event = { id = vie_infra.3 days = 3 random_days = 20 }
		}
		else_if = {
			limit = { date > 2020.9.30 }
			set_country_flag = VIE_sched_vna_covid
			VIE_fb_none = yes
		}
	}
}
