			limit = {
				check_variable = {
					VIE_infra_road_foundation_stage = 0

			}
			set_temp_variable = {
				skip_payment = 1
			}

			}

			}
			set_temp_variable = {
				skip_payment = 0
			}

				VIE_infra_road_foundation_stage = 1
			}

				VIE_expressway_km = 400
				tooltip = VIE_expressway_km_tt
			}

				var = VIE_expressway_km

				max = 5000
