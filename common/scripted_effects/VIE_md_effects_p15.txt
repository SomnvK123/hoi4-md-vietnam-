			limit = {
				date > 2022.11.30
				NOT = { has_country_flag = VIE_def_ind_sched_1_done }
			}
			if = {
				limit = {
					VIE_def_ind_level_ge_2 = yes
					NOT = { has_country_flag = VIE_popup_cd }
				}
				set_country_flag = VIE_def_ind_sched_1_done

				country_event = { id = vie_def_ind.1 days = 3 random_days = 15 }
			}

				limit = { date > 2023.6.30 }
				set_country_flag = VIE_def_ind_sched_1_done
				VIE_gdt_mio_funds_100 = yes
			}
		}
		if = {
			limit = {
				date > 2024.11.30
				NOT = { has_country_flag = VIE_def_ind_sched_2_done }
			}
			if = {
				limit = {
					VIE_def_ind_level_ge_2 = yes
					NOT = { has_country_flag = VIE_popup_cd }
				}
				set_country_flag = VIE_def_ind_sched_2_done
				set_country_flag = { flag = VIE_popup_cd value = 1 days = 45 }
				country_event = { id = vie_def_ind.1 days = 3 random_days = 15 }

				limit = { date > 2025.6.30 }
				set_country_flag = VIE_def_ind_sched_2_done
				VIE_gdt_mio_funds_100 = yes
			}
		}

		# vie_def_ind.2 - Chuyen giao bao duong T-90.
		# Can VIE_t90_purchased (Truc 1 .15 da dat) va D1 xong.
		#
		# KHOI NAY CO Y KHONG CO FALLBACK va khong co han chot, khac voi .1 va .4.
