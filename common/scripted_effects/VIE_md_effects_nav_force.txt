# Decision D-A: huan luyen luc luong tau mat nuoc (event vie_nav_force.1 .2, ket thuc .61).
# Nguoi goi dat truoc VIE_nf_surface_level (1 co ban / 2 chuyen sau) va VIE_nf_surface_specialty
# (1 hoa luc / 2 chong ngam). Chi phi 0,40 / 0,60 ty, thoi luong 365 / 548 ngay.
########################################################################
VIE_nf_d1_start = {
	# Bien tam dat truoc (mac dinh muc 1) roi ghi de trong if: bien tam dat ben trong if khong dung duoc o ngoai.
	set_temp_variable = { treasury_change = -0.4 }
	set_temp_variable = { VIE_nf_end_days = 365 }
	if = {
		limit = { check_variable = { VIE_nf_surface_level = 2 } }
		set_temp_variable = { treasury_change = -0.6 }
		set_temp_variable = { VIE_nf_end_days = 548 }
	}
	modify_treasury_effect = yes
	add_timed_idea = { idea = VIE_nf_prog_surface days = VIE_nf_end_days }
	country_event = { id = vie_nav_force.61 days = VIE_nf_end_days }
}

# Ket thuc D-A. Muc 2 = 1,5 lan muc 1.
VIE_nf_d1_finish = {
	if = {
		limit = { check_variable = { VIE_nf_surface_specialty = 2 } }
		if = {
			limit = { check_variable = { VIE_nf_surface_level = 2 } }
			add_to_variable = { VIE_af_navy_submarine_defence_factor = 0.045 tooltip = VIE_tt_navy_submarine_defence_factor }
			add_to_variable = { VIE_af_naval_detection = 0.03 tooltip = VIE_tt_naval_detection }
		}
		else = {
			add_to_variable = { VIE_af_navy_submarine_defence_factor = 0.03 tooltip = VIE_tt_navy_submarine_defence_factor }
			add_to_variable = { VIE_af_naval_detection = 0.02 tooltip = VIE_tt_naval_detection }
		}
	}
	else = {
		if = {
			limit = { check_variable = { VIE_nf_surface_level = 2 } }
			add_to_variable = { VIE_af_naval_strike_attack_factor = 0.045 tooltip = VIE_tt_naval_strike_attack_factor }
			add_to_variable = { VIE_af_navy_org_factor = 0.015 tooltip = VIE_tt_navy_org_factor }
		}
		else = {
			add_to_variable = { VIE_af_naval_strike_attack_factor = 0.03 tooltip = VIE_tt_naval_strike_attack_factor }
			add_to_variable = { VIE_af_navy_org_factor = 0.01 tooltip = VIE_tt_navy_org_factor }
		}
	}
	VIE_nf_dm_tt = yes
	set_country_flag = VIE_nf_d1_done
	VIE_nf_program_end = yes
}
