# Scheduler, part 11 (plan v6 batch 7): the last conditional events. Skipped in catch-up mode.
VIE_event_scheduler_p11 = {
	if = {
		limit = { NOT = { check_variable = { VIE_catch_up = 1 } } }

		# Door D7: a legitimacy crisis after the pandemic years, if stability is low
		if = {
			limit = {
				VIE_party_rule_active = yes
				date > 2021.8.31
				has_stability < 0.35
			}
			if = {
				limit = { NOT = { has_country_flag = VIE_door_d7_done } }
				set_country_flag = VIE_door_d7_done
				set_country_flag = { flag = VIE_popup_cd value = 1 days = 45 }
				country_event = { id = vie_alt.23 days = 20 random_days = 20 }
			}
		}

		# South China Sea: level 3 (a clash at the Paracels), once
		if = {
			limit = {
				check_variable = { VIE_scs_tension = 3 }
				NOT = { has_country_flag = VIE_paracel_clash_seen }
				NOT = { has_country_flag = VIE_popup_cd }
			}
			set_country_flag = VIE_paracel_clash_seen
			set_country_flag = { flag = VIE_popup_cd value = 1 days = 45 }
			country_event = { id = vie_scs.16 days = 3 }
		}

		# ...and ceasefire talks once that war has run for four months
		if = {
			limit = {
				has_war_with = CHI
				has_country_flag = VIE_paracel_clash
				NOT = { has_country_flag = VIE_war_clock }
				NOT = { has_country_flag = VIE_truce_talks_done }
			}
			set_country_flag = VIE_truce_talks_done
			country_event = { id = vie_scs.17 days = 3 }
		}

		# ASEAN mediation during the civil war, once
		if = {
			limit = {
				has_civil_war = yes
				has_country_flag = VIE_collapse_done
				NOT = { has_country_flag = VIE_asean_mediation_offered }
			}
			random = {
				chance = 20
				set_country_flag = VIE_asean_mediation_offered
				country_event = { id = vie_col.8 days = 5 }
			}
		}
	}
}
