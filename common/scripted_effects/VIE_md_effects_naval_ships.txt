# Gepard I: cua so 2006-01 -> 2008-12. Event .10 (cau hinh) -> .11 (so luong).
VIE_naval_sched_gepard1 = {
	if = {
		limit = {
			NOT = { has_country_flag = VIE_gepard1_offered }
			NOT = { has_country_flag = VIE_gepard1_missed }
		}
		if = {
			limit = {
				date > 2005.12.31
				date < 2009.1.1
				country_exists = SOV
				NOT = { has_war_with = SOV }
			}
			if = {
				limit = { NOT = { has_country_flag = VIE_popup_cd } }
				set_country_flag = VIE_gepard1_offered
				set_country_flag = { flag = VIE_popup_cd value = 1 days = 45 }
				country_event = { id = vie_naval.10 days = 1 random_days = 14 }
			}
			else = {
				set_country_flag = VIE_gepard1_gate_seen
			}
		}
		else_if = {
			limit = { date > 2008.12.31 }
			if = {
				limit = { has_country_flag = VIE_gepard1_gate_seen }
				set_country_flag = VIE_gepard1_offered
				VIE_fb_naval_gepard1 = yes
			}
			else = {
				set_country_flag = VIE_gepard1_missed
			}
		}
	}
}

# Kilo: 0,333 moi tau cho than tau + 0,2 moi tau neu huan luyen/ha tang day du.
VIE_naval_pay_kilo = {
	set_temp_variable = { treasury_change = VIE_kilo_qty_ordered }
	multiply_temp_variable = { treasury_change = -0.3333 }
	modify_treasury_effect = yes
	if = {
		limit = { has_country_flag = VIE_kilo_training_full }
		set_temp_variable = { treasury_change = VIE_kilo_qty_ordered }
		# Truc 3 (L7): da chuan bi nhan luc tau ngam thi goi huan luyen re hon (0,15 thay 0,2 ty/tau).
		# Chi la thuong, khong phai dieu kien (Quy tac 5).
		if = {
			limit = { has_country_flag = VIE_nf_sub_prep }
			multiply_temp_variable = { treasury_change = -0.15 }
		}
		else = {
			multiply_temp_variable = { treasury_change = -0.2 }
		}
		modify_treasury_effect = yes
		navy_experience = 20
	}
}
