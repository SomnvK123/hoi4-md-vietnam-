# Khoi tao toan bo he thong chinh tri
VIE_init_political_system = {
	if = {
		limit = { NOT = { has_country_flag = VIE_political_system_initialized } }
		set_country_flag = VIE_political_system_initialized
		set_variable = { VIE_leadership_cohesion = 60 }
		set_variable = { VIE_last_congress = 8 } # Truoc Dai hoi IX (2001)
		set_variable = { VIE_tier_cadre = 0 }
		set_variable = { VIE_tier_admin = 0 }
		set_variable = { VIE_tier_inspection = 0 }
		set_variable = { VIE_tier_social = 0 }
		set_variable = { VIE_programme_epoch = 0 }
		set_variable = { VIE_verification_counter = 0 }
		set_variable = { VIE_office_holders_dual_role = 0 }
		set_power_balance = {
			id = VIE_leadership_balance
			left_side = VIE_bop_side_collective
			right_side = VIE_bop_side_central
			set_value = 0.0
		}
	}
}

# Cap nhat Cohesion co clamp 0..100
# Caller: set_temp_variable = { cohesion_add = N }
VIE_modify_cohesion = {
	custom_effect_tooltip = VIE_tt_modify_cohesion
	hidden_effect = {
		add_to_variable = { VIE_leadership_cohesion = cohesion_add }
		clamp_variable = { var = VIE_leadership_cohesion min = 0 max = 100 }
	}
}

VIE_launch_programme_standard = {
	log = "[GetDateText]: [Root.GetName]: Launch programme standard"
	set_country_flag = VIE_programme_active
	clr_country_flag = VIE_programme_selected
	set_country_flag = VIE_mode_standard
	clr_country_flag = VIE_mode_fast_paid
	clr_country_flag = VIE_mode_fast_override
	set_variable = { VIE_programme_targets_met = 0 }
	set_variable = { VIE_verification_counter = 0 }

	# Chi phi khoi dong tieu chuan: hoan tra PP dua tren Bậc cán bộ (base 40 da tru boi decision cost)
	set_temp_variable = { startup_discount = 0 }
	if = { limit = { check_variable = { VIE_tier_cadre > 2 } } add_to_temp_variable = { startup_discount = 15 } }
	else_if = { limit = { check_variable = { VIE_tier_cadre > 1 } } add_to_temp_variable = { startup_discount = 10 } }
	else_if = { limit = { check_variable = { VIE_tier_cadre > 0 } } add_to_temp_variable = { startup_discount = 5 } }

	if = {
		limit = { check_variable = { startup_discount > 0 } }
		add_political_power = startup_discount
	}
}

# Resolver Chuong trinh (Full / Partial / Fail)
# ------------------------------------------------------------
VIE_resolve_programme_full = {
	log = "[GetDateText]: [Root.GetName]: Resolve programme FULL"
	set_country_flag = VIE_last_programme_full
	clr_country_flag = VIE_programme_active
	set_temp_variable = { cohesion_add = 6 }
	VIE_modify_cohesion = yes

	# Thuong them tap the neu dang o vung Collective (toi da +4, tran 70)
	if = {
		limit = {
			VIE_is_collective_zone = yes
			check_variable = { VIE_leadership_cohesion < 70 }
		}
		set_temp_variable = { cohesion_add = 4 }
		VIE_modify_cohesion = yes
	}

	# Thuong som 50 PP neu hoan thanh truoc moc va khong bi mat quyen
	if = {
		limit = {
			NOT = { has_country_flag = VIE_e103_early_reward_forfeited }
		}
		add_political_power = 50
	}

	# Tang Tier tuong ung & thuong chuyen nganh
	if = {
		limit = { has_country_flag = VIE_programme_cadre }
		VIE_award_tier_cadre = yes
		add_political_power = 40
	}
	if = {
		limit = { has_country_flag = VIE_programme_admin }
		VIE_award_tier_admin = yes
		increase_economic_growth = yes
	}
	if = {
		limit = { has_country_flag = VIE_programme_inspection }
		VIE_award_tier_inspection = yes
		decrease_corruption = yes
	}
	if = {
		limit = { has_country_flag = VIE_programme_social }
		VIE_award_tier_social = yes
		add_stability = 0.03
	}
}

VIE_award_tier_cadre = {
	if = {
		limit = { check_variable = { VIE_tier_cadre < 1 } }
		set_variable = { VIE_tier_cadre = 1 }
		add_ideas = VIE_tier_cadre_1
	}
	else_if = {
		limit = { check_variable = { VIE_tier_cadre < 2 } }
		set_variable = { VIE_tier_cadre = 2 }
		swap_ideas = { remove_idea = VIE_tier_cadre_1 add_idea = VIE_tier_cadre_2 }
	}
	else_if = {
		limit = { check_variable = { VIE_tier_cadre < 3 } }
		set_variable = { VIE_tier_cadre = 3 }
		swap_ideas = { remove_idea = VIE_tier_cadre_2 add_idea = VIE_tier_cadre_3 }
	}
}

VIE_apply_congress_transition_cohesion = {
	# Ton that cohesion khi chuyen giao (tier can bo giam ton that: 12 -> 8 -> 5 -> 3)
	if = { limit = { check_variable = { VIE_tier_cadre > 2 } } set_temp_variable = { cohesion_add = -3 } }
	else_if = { limit = { check_variable = { VIE_tier_cadre > 1 } } set_temp_variable = { cohesion_add = -5 } }
	else_if = { limit = { check_variable = { VIE_tier_cadre > 0 } } set_temp_variable = { cohesion_add = -8 } }
	else = { set_temp_variable = { cohesion_add = -12 } }

	# Phu phi neu dang giu kiem nhiem
	if = {
		limit = { has_country_flag = VIE_dual_role_active }
		subtract_from_temp_variable = { cohesion_add = 4 }
	}
	VIE_modify_cohesion = yes
}
