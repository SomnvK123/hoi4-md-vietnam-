# Vietnam submod scripted effects (Phase 3): doors, regime ticks, collapse and civil war.
# All of it is called from VIE_event_scheduler_p3 (monthly) or from focuses/events.

########################################################################
# Regime entry helper: idempotent wrapper around VIE_transition_regime with a popularity boost.
#   VIE_enter_regime = { PARTY = 20 AMOUNT = 0.20 }
########################################################################
########################################################################
# Doors: a conditional event that fires once; the event's own options gate the alternatives.
########################################################################
########################################################################
# Re-creates the "already happened" flags in catch-up mode (VIE_catch_up = 1). Not called by any hook now; kept for later use.
# The winner may be the old country or the rebel country, so it re-creates the "already happened" flags.
########################################################################
VIE_catch_up_schedule = {
	set_temp_variable = { VIE_catch_up = 1 }
	VIE_event_scheduler = yes
	VIE_event_scheduler_p2 = yes
	VIE_event_scheduler_p3 = yes
	VIE_event_scheduler_p5 = yes
	VIE_event_scheduler_p6 = yes
	VIE_event_scheduler_p7 = yes
	VIE_event_scheduler_p10 = yes
	VIE_event_scheduler_p13 = yes
	VIE_event_scheduler_p14 = yes
	VIE_event_scheduler_p15 = yes
	VIE_event_scheduler_p17 = yes
}

########################################################################
# Silent fallbacks / small helpers used by regime events
########################################################################
########################################################################
# Scheduler, part 3.  Called monthly after part 1 and part 2.  Skipped in catch-up mode.
########################################################################
VIE_event_scheduler_p3 = {
	if = {
		limit = { NOT = { check_variable = { VIE_catch_up = 1 } } }

		# ---- Doors (fire once each; options are gated inside the events)
		if = {
			limit = {
				VIE_party_rule_active = yes
				has_country_flag = VIE_hd981_done
				NOT = { has_country_flag = VIE_regime_changed_recently }
			}
			if = {
				limit = { NOT = { has_country_flag = VIE_door_d4_done } }
				hidden_effect = { set_country_flag = VIE_door_d4_done }
				hidden_effect = { set_country_flag = { flag = VIE_popup_cd value = 1 days = 45 } }
				country_event = { id = vie_alt.1 days = 20 random_days = 20 }
			}
		}
		if = {
			limit = {
				VIE_party_rule_active = yes
				date > 2006.10.31
				has_country_flag = VIE_press_relaxed
			}
			if = {
				limit = { NOT = { has_country_flag = VIE_door_d2_done } }
				hidden_effect = { set_country_flag = VIE_door_d2_done }
				hidden_effect = { set_country_flag = { flag = VIE_popup_cd value = 1 days = 45 } }
				country_event = { id = vie_alt.2 days = 20 random_days = 20 }
			}
		}
		if = {
			limit = {
				VIE_party_rule_active = yes
				date > 2011.5.31
				OR = {
					has_stability < 0.55
					check_variable = { VIE_vinashin_risk > 1 }
				}
			}
			if = {
				limit = { NOT = { has_country_flag = VIE_door_d3_done } }
				hidden_effect = { set_country_flag = VIE_door_d3_done }
				hidden_effect = { set_country_flag = { flag = VIE_popup_cd value = 1 days = 45 } }
				country_event = { id = vie_alt.3 days = 20 random_days = 20 }
			}
		}
		if = {
			limit = {
				VIE_party_rule_active = yes
				date > 2018.5.31
			}
			if = {
				limit = { NOT = { has_country_flag = VIE_door_d6_done } }
				hidden_effect = { set_country_flag = VIE_door_d6_done }
				hidden_effect = { set_country_flag = { flag = VIE_popup_cd value = 1 days = 45 } }
				country_event = { id = vie_alt.6 days = 20 random_days = 20 }
			}
		}
		if = {
			limit = {
				VIE_party_rule_active = yes
				date > 2022.9.30
			}
			if = {
				limit = { NOT = { has_country_flag = VIE_door_d8_done } }
				hidden_effect = { set_country_flag = VIE_door_d8_done }
				hidden_effect = { set_country_flag = { flag = VIE_popup_cd value = 1 days = 45 } }
				country_event = { id = vie_alt.7 days = 20 random_days = 20 }
			}
		}

	}
}
