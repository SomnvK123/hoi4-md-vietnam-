		limit = {

		}
		hidden_effect = { set_country_flag = VIE_sched_petro_crisis }
		country_event = { id = vie_petro.1 days = 90 }
	}

	# Lien minh xe dien: chi len lich sau khi ca hai focus da hoan tat.

		}

	}
