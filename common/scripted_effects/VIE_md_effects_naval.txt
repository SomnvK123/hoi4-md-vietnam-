# Chu y: _gate_seen nho rang cong tung dat trong cua so ma popup bi chan; het cua so thi
# ap ket qua lich su im lang thay vi _missed (bao cao review muc B5).
########################################################################
VIE_naval_sched_molniya_p1 = {
	if = {
		limit = {
			NOT = { has_country_flag = VIE_molniya_p1_offered }
			NOT = { has_country_flag = VIE_molniya_p1_missed }
		}
		if = {
			limit = {
				date > 2003.5.31
				date < 2006.1.1
				country_exists = SOV
				NOT = { has_war_with = SOV }
			}
			if = {
				limit = { NOT = { has_country_flag = VIE_popup_cd } }
				set_country_flag = VIE_molniya_p1_offered
				set_country_flag = { flag = VIE_popup_cd value = 1 days = 45 }
				country_event = { id = vie_naval.1 days = 1 random_days = 14 }
			}
			else = {
				set_country_flag = VIE_molniya_p1_gate_seen
			}
		}
		else_if = {
			limit = { date > 2005.12.31 }
			if = {
				limit = { has_country_flag = VIE_molniya_p1_gate_seen }
				set_country_flag = VIE_molniya_p1_offered
				VIE_fb_naval_molniya_p1 = yes
			}
			else = {
				set_country_flag = VIE_molniya_p1_missed
			}
		}
	}
}
