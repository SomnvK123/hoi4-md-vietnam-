# ---- Decision 1: khoi dong chuong trinh Ba Son ----
# Nguoi goi (vie_nav_ind.2 option) dat truoc cac bien tam:
#   VIE_nav_invest   1 Co ban / 2 Mo rong / 3 Trong diem
#   VIE_nav_total    tong gia theo bac, ty USD (7,5 / 12 / 18); dockyard do MD tu tru, phan con
#   VIE_nav_hand     lai tru tay (0 / 4,5 / 3)
#   VIE_nav_ms_days  moc 40% (146 / 219 / 292)
#   VIE_nav_end_days ket thuc (365 / 548 / 730)
# Dinh huong Can bang (VIE_ba_son_orientation = 3) chiu them 15% tong gia.
# Moc 40% dat VIE_cap_ba_son_yard va VIE_var_ba_son_tier ngay (review Truc 2, A4): Truc 1 doc ca hai
# o cua so Molniya pha 2; dockyard, MIO va kinh nghiem tra khi ket thuc.
VIE_nav_d1_start = {
	set_variable = { VIE_ba_son_invest = VIE_nav_invest }
	if = {
		limit = { check_variable = { VIE_nav_hand > 0 } }
		set_temp_variable = { treasury_change = VIE_nav_hand }
		multiply_temp_variable = { treasury_change = -1 }
		modify_treasury_effect = yes
	}
	if = {
		limit = { check_variable = { VIE_ba_son_orientation = 3 } }
		set_temp_variable = { treasury_change = VIE_nav_total }
		multiply_temp_variable = { treasury_change = -0.15 }
		modify_treasury_effect = yes
	}
	add_timed_idea = { idea = VIE_nav_prog_ba_son days = VIE_nav_end_days }
	country_event = { id = vie_nav_ind.60 days = VIE_nav_ms_days }
	country_event = { id = vie_nav_ind.61 days = VIE_nav_end_days }
}

# Ket thuc: dockyard xay o state 519 (TP Ho Chi Minh, noi co Ba Son). one_state_dockyard va

	else_if = {
		limit = { check_variable = { VIE_ba_son_invest = 2 } }
		519 = { one_state_dockyard = yes }
		VIE_ba_son_mio_size_2 = yes
		set_temp_variable = { VIE_nav_exp_gain = 10 }
		VIE_nav_add_shipbuilding_exp = yes
	}
	else = {
		519 = { one_state_dockyard = yes }
		VIE_ba_son_mio_size_1 = yes
		set_temp_variable = { VIE_nav_exp_gain = 5 }
		VIE_nav_add_shipbuilding_exp = yes
	}
	hidden_effect = { set_country_flag = VIE_nav_d1_done }
	VIE_nav_program_end = yes
}


# ---- Decision 2 (MRO) ----
# Nguoi goi (vie_nav_ind.12) dat truoc bien tam: VIE_nav_level (1-3), VIE_nav_base_days (365/548/730),
# VIE_nav_cost (4,0 / 6,4 / 9,6 = goc 4,0 x 1,0 / 1,6 / 2,4). Bien luu: VIE_mro_scope (1 tau ngam,
# 2 tau mat nuoc, 3 toan ham doi), VIE_mro_source (1 Nga, 2 tu chu).
# Chi phi = goc x (0,8 Nga | 1,4 tu chu) x (1,3 neu toan ham doi). Thoi luong = goc x (0,75 Nga | 1,25 tu chu)
# x (0,75 dinh huong MRO truoc | 1,25 dong tau truoc) x (0,85 neu VIE_opp_sub_mro va co tau ngam), lam tron.
VIE_nav_d2_start = {
	set_variable = { VIE_mro_level = VIE_nav_level }
	if = {
		limit = { check_variable = { VIE_mro_source = 1 } }
		multiply_temp_variable = { VIE_nav_cost = 0.8 }
	}
	else = {
		multiply_temp_variable = { VIE_nav_cost = 1.4 }
	}
	if = {
		limit = { check_variable = { VIE_mro_scope = 3 } }
		multiply_temp_variable = { VIE_nav_cost = 1.3 }
	}
	set_temp_variable = { treasury_change = VIE_nav_cost }
	multiply_temp_variable = { treasury_change = -1 }
	modify_treasury_effect = yes

	set_temp_variable = { VIE_nav_end_days = VIE_nav_base_days }
	if = {
		limit = { check_variable = { VIE_mro_source = 1 } }
		multiply_temp_variable = { VIE_nav_end_days = 0.75 }
	}
	else = {
		multiply_temp_variable = { VIE_nav_end_days = 1.25 }
	}
	if = {
		limit = { check_variable = { VIE_ba_son_orientation = 2 } }
		multiply_temp_variable = { VIE_nav_end_days = 0.75 }
	}
	else_if = {
		limit = { check_variable = { VIE_ba_son_orientation = 1 } }
		multiply_temp_variable = { VIE_nav_end_days = 1.25 }
	}
	if = {
		limit = {
			has_country_flag = VIE_opp_sub_mro
			NOT = { check_variable = { VIE_mro_scope = 2 } }
		}
		multiply_temp_variable = { VIE_nav_end_days = 0.85 }
	}
	round_temp_variable = VIE_nav_end_days
	add_timed_idea = { idea = VIE_nav_prog_mro days = VIE_nav_end_days }
	country_event = { id = vie_nav_ind.62 days = VIE_nav_end_days }
}
