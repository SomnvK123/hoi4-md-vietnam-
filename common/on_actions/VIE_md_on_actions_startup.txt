on_actions = {
	# Translates the VIE_ai_behavior game rule into a global flag (pattern of MD's 999_game_rules_on_actions.txt).
	# The flags are read by VIE_ai_path_* triggers, the focus ai_will_do blocks and the event ai_chance blocks.
	on_startup = {
		effect = {
			VIE = {
				# 2000-2014 army roster (see VIE_land_forces_implementation_plan_v2.md). on_startup also runs on
				# every save load, so the flag keeps this from recruiting twice; vie_army_commanders.2 retires
				# the group on 2015.1.1. VIE_army_le_manh is defined but left out on purpose (6 commander cap).
				if = {
					limit = {
						date < 2015.1.1
						NOT = { has_country_flag = VIE_army_phase1_recruited }
					}
					recruit_character = VIE_army_le_van_dung
					recruit_character = VIE_army_phung_quang_thanh
					recruit_character = VIE_army_huynh_tien_phong
					recruit_character = VIE_army_nguyen_khac_nghien
					recruit_character = VIE_army_do_ba_ty
					recruit_character = VIE_army_pham_van_tra
					recruit_character = VIE_army_nguyen_van_duoc
					recruit_character = VIE_army_hoang_ky
					recruit_character = VIE_army_pham_xuan_hung
					set_country_flag = VIE_army_phase1_recruited
				}
				# Air force step 0 (later steps: VIE_event_scheduler_air). The two upstream air chiefs are not air
				# officers (a political general and an army/academy general), so they are retired, not replaced.
				# Same on_startup caveat as above: the flag stops a reload from recruiting and retiring again.
				if = {
					limit = {
						date < 2002.2.7
						NOT = { has_country_flag = VIE_air_phase0_done }
					}
					recruit_character = VIE_air_nguyen_duc_soat
					recruit_character = VIE_air_pham_thanh_ngan
					recruit_character = VIE_air_han_vinh_tuong
					recruit_character = VIE_air_pham_tuan
					recruit_character = VIE_air_nguyen_van_phiet
					retire_character = VIE_Tran_Quang_Phuong
					retire_character = VIE_Tran_Viet_Khoa
					set_country_flag = VIE_air_phase0_done
				}
				if = {
					limit = { has_game_rule = { rule = VIE_ai_behavior option = VIE_HISTORICAL } }
					set_global_flag = VIE_AI_PATH_HISTORICAL
				}
				if = {
					limit = { has_game_rule = { rule = VIE_ai_behavior option = VIE_REFORM } }
					set_global_flag = VIE_AI_PATH_REFORM
				}
				if = {
					limit = { has_game_rule = { rule = VIE_ai_behavior option = VIE_HARDLINE } }
					set_global_flag = VIE_AI_PATH_HARDLINE
				}
				if = {
					limit = { has_game_rule = { rule = VIE_ai_behavior option = VIE_WESTERN } }
					set_global_flag = VIE_AI_PATH_WESTERN
				}
				if = {
					limit = { has_game_rule = { rule = VIE_ai_behavior option = RANDOM } }
					random_list = {
						20 = { set_global_flag = VIE_AI_PATH_HISTORICAL }
						15 = { set_global_flag = VIE_AI_PATH_REFORM }
						10 = { set_global_flag = VIE_AI_PATH_HARDLINE }
						10 = { set_global_flag = VIE_AI_PATH_WESTERN }
						45 = { }
					}
				}
				# Initialise the 9-axis state-building system at game start,
				# before the first on_monthly tick fires, to prevent stack overflow.
				VIE_ax_init = yes

				# Truc 2 (CNQP Luc quan): bien level. HOI4 coi bien chua dat = 0 nen ve
				# ky thuat khong bat buoc, nhung VIE_nationalist_regime_TDD.md muc 7.3 cua
				# chinh repo nay da nhac: "Dat ro ... = 0, du bien chua dat mac dinh bang 0".
				# Dat tu dau de check_variable >= 4 / >= 8 khong bao gio doc gia tri rac,
				# va de save cu tai lai van dung.
				set_variable = { VIE_def_industry_level = 0 }
				set_variable = { VIE_def_ind_export_count = 0 }

				# Party Congress spine (see VIE_congress_spine_redesign.md): saves from before
				# this system existed have no VIE_congress_term variable. Re-derive it every
				# startup from the scheduler flags already set - cheap, and self-correcting.
				set_variable = { VIE_congress_term = 8 }
				if = { limit = { has_country_flag = VIE_sched_congress_9 } set_variable = { VIE_congress_term = 9 } }
				if = { limit = { has_country_flag = VIE_sched_congress_10 } set_variable = { VIE_congress_term = 10 } }
				if = { limit = { has_country_flag = VIE_sched_congress_11 } set_variable = { VIE_congress_term = 11 } }
				if = { limit = { has_country_flag = VIE_sched_congress_12 } set_variable = { VIE_congress_term = 12 } }
				if = { limit = { has_country_flag = VIE_sched_congress_13 } set_variable = { VIE_congress_term = 13 } }
				if = { limit = { has_country_flag = VIE_sched_congress_14 } set_variable = { VIE_congress_term = 14 } }

				# VIE_two_tier_local_gov was deleted in Phase 3 (section 8.3): a save that had
				# already completed it kept the idea it granted but lost the flag the new
				# 2026 fork and VIE_era_of_rising now gate on. Re-derive that flag from the idea.
				if = {
					limit = { has_idea = VIE_two_tier_gov_idea }
					set_country_flag = VIE_two_tier_done
				}
			}
		}
	}
}
