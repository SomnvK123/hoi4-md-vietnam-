# Historical force mix: favor light infantry and reduce expensive mechanized/armor roles.
# The role IDs are supplied by MD's generic ai_templates; these are ratio adjustments,
# not hard caps or a replacement for MD's production logic.
VIE_historical_force_mix = {
	allowed = {
		original_tag = VIE
		ai_for_units_is_not_disabled = yes
	}
	enable = {
		VIE_ai_historical = yes
		NOT = { has_completed_focus = VIE_lf_m1_combined_arms_mechanized }
		NOT = { has_completed_focus = VIE_lf_r1_flexible_mobile_warfare }
	}
	abort_when_not_enabled = yes

	ai_strategy = { type = role_ratio id = L_Inf value = 25 }
	ai_strategy = { type = role_ratio id = infantry value = 10 }
	ai_strategy = { type = role_ratio id = Militia value = 2 }
	ai_strategy = { type = role_ratio id = apc_mechanized value = -20 }
	ai_strategy = { type = role_ratio id = ifv_mechanized value = -20 }
	ai_strategy = { type = role_ratio id = armor value = -20 }
	ai_strategy = { type = role_ratio id = marines value = -5 }
	ai_strategy = { type = role_ratio id = Air_mech value = -3 }
	ai_strategy = { type = role_ratio id = Special_Forces value = -5 }
}
