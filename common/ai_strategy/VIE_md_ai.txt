# AI strategies for an AI-controlled Vietnam (submod). MD had no VIE file. Patterns follow SIN.txt / SIA.txt.
# The VIE_path_* blocks are switched on by the game rule "Vietnam AI behaviour" (global flags VIE_AI_PATH_*).

# ---- Default: bamboo diplomacy. Friends in ASEAN and among the big partners, no war with anyone.
VIE_befriend_asean = {
	allowed = { original_tag = VIE }
	enable = {
		NOT = { has_government = nationalist }
	}
	abort_when_not_enabled = yes

	ai_strategy = { type = befriend id = "SIA" value = 100 }
	ai_strategy = { type = befriend id = "LAO" value = 150 }
	ai_strategy = { type = befriend id = "CBD" value = 100 }
	ai_strategy = { type = befriend id = "SIN" value = 100 }
	ai_strategy = { type = befriend id = "MAY" value = 100 }
	ai_strategy = { type = befriend id = "IND" value = 100 }
	ai_strategy = { type = befriend id = "PHI" value = 50 }

}

# Vietnam never starts a war on China, and joins no bloc: the Four Nos also block factions through an idea.
VIE_avoid_war_with_china = {
	allowed = { original_tag = VIE }
	enable = {
		NOT = { has_global_flag = VIE_AI_PATH_NATIONALIST }
	}
	abort_when_not_enabled = yes

	ai_strategy = { type = declare_war id = "CHI" value = -300 }
}
