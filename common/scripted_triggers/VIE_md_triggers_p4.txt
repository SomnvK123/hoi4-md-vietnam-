# Vietnam AI behaviour game rule. Historical remains the locked-by-default choice set;
# Hardline selectively enables only the AI choices explicitly whitelisted for this path.

# Keep the broad historical-choice kill switch permanent. Hardline exceptions use VIE_ai_hardline
# directly so selecting this path cannot unlock unrelated alternative-history choices.
VIE_ai_historical = {
	always = yes
}

# Selected AI profile; separate from VIE_ai_historical, which remains the broad
# historical-choice lock for whitelisted and non-whitelisted alternative choices.

	has_global_flag = VIE_AI_PATH_HISTORICAL
}

VIE_ai_hardline = {
	has_global_flag = VIE_AI_PATH_HARDLINE
}

# Decisions marked "AI only when VIE_ai_free" stay off for the AI (historical line is permanent).
VIE_ai_free = {
	always = no
}

VIE_gate_security_stability = {
