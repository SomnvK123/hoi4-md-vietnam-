# Doi tac $T$ co the gui tau tham Cam Ranh: ton tai, khong co chien tranh voi VIE, chua tham trong 3 nam qua.
VIE_scs_pc_ok = {
	country_exists = $T$
	NOT = { has_war_with = $T$ }
	NOT = { has_country_flag = VIE_scs_pc_$T$ }
}

# It nhat mot doi tac co the tham cang (cong cua decision VIE_scs_d2_port_call).
