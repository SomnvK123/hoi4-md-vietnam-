}

}

	check_variable = { VIE_apm_uav_tier > 1 }
}

VIE_airf_capstone_ok = {
	OR = {
		AND = { check_variable = { VIE_airf_capstone_target = 1 } has_completed_focus = VIE_airf_iads_command VIE_airf_capstone_ok_a = yes }
		AND = { check_variable = { VIE_airf_capstone_target = 2 } has_completed_focus = VIE_airf_multirole_wing VIE_airf_capstone_ok_b = yes }
		AND = { check_variable = { VIE_airf_capstone_target = 3 } has_completed_focus = VIE_airf_teaming VIE_airf_capstone_ok_c = yes }
	}
}
