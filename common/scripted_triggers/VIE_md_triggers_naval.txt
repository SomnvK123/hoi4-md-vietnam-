# ---------------------------------------------------------------------
# Funding Gate (Sigma, va cac chuong trinh khac neu muon chan theo ngan kho).
# MD luu ngan kho o bien `treasury` (don vi ty USD; modify_treasury_effect cong vao
# bien nay, xem 00_budget_effects.txt). Khong dung bien sao chep naval_budget_room.
# Nguoi goi phai dat truoc: set_temp_variable = { VIE_naval_cost = <ty USD> }
#
# Khac convention Truc 2 (thieu tien khong chan decision vi MD tu phat hanh no):
# day la ngoai le co chu y, cong xac dinh cua Sigma. Tooltip phai noi ro nguong.
# ---------------------------------------------------------------------
VIE_naval_can_fund = {
	check_variable = { treasury > VIE_naval_cost }
}

# ---------------------------------------------------------------------
# Gepard II chi mo khi Gepard I da giao xong (ngoai le "chuong trinh noi tiep", bao cao 3.1).
# So sanh bien-voi-bien bang NOT(<) de tranh phu thuoc toan tu >=.
# ---------------------------------------------------------------------
VIE_naval_gepard1_done = {
	has_country_flag = VIE_gepard1_contracted
	check_variable = { VIE_gepard1_qty_ordered > 0 }
	NOT = { check_variable = { VIE_gepard1_qty_delivered < VIE_gepard1_qty_ordered } }
}

# ---------------------------------------------------------------------
# Truc 2 hai quan: dieu kien doc tu Truc 1 (VIE_md_effects_naval_ships.txt, giao hang) va nguon
# nang luc VIE_ext_* (plan VIE_naval_truc2_review_and_plan.md muc 7). Gom o day de doi mot dong
# khi nhanh Viettel/C4ISR that duoc xay.
# ---------------------------------------------------------------------
VIE_naval_has_sub = {
	check_variable = { VIE_kilo_qty_delivered > 0 }
}

VIE_naval_has_surface = {
	OR = {
		check_variable = { VIE_gepard1_qty_delivered > 0 }
		check_variable = { VIE_gepard2_qty_delivered > 0 }
		check_variable = { VIE_molniya_ru_delivered > 0 }
		check_variable = { VIE_molniya_vn_qty_delivered > 0 }
	}
}

# Dien tu: ban dan dan dung hoa cua dan su (dual-use). TODO: doi khi co nhanh Viettel quan su.
VIE_naval_has_electronics = {
	OR = {
		has_completed_focus = VIE_semiconductor_fab
		has_completed_focus = VIE_chip_design
	}
}

# C4ISR: ve tinh quan sat / vien thong. TODO: doi khi co nhanh C4ISR.
VIE_naval_has_c4isr = {
	OR = {
		has_completed_focus = VIE_earth_observation
		has_completed_focus = VIE_vinasat
	}
}

# Ten lua hai quan: Truc 1 dat co khi giao du Bastion-P.
VIE_naval_has_missile = {
	has_country_flag = VIE_ext_naval_missile
}

# Mot trong ba nguon nang luc (Focus 5).
VIE_naval_has_ext_source = {
	OR = {
		VIE_naval_has_electronics = yes
		VIE_naval_has_c4isr = yes
		VIE_naval_has_missile = yes
	}
}
