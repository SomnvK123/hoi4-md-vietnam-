# One of the "poles" that turns a crisis into a collapse.
VIE_collapse_pole = {
	OR = {
		AND = {
			has_country_flag = VIE_bop_oli_active
			power_balance_value = { id = VIE_oligarch_balance value < -0.85 }
		}
		AND = {
			has_country_flag = VIE_bop_active
			OR = {
				power_balance_value = { id = VIE_party_balance value < -0.85 }
				power_balance_value = { id = VIE_party_balance value > 0.85 }
			}
		}
		has_country_flag = VIE_wa_suppressed
		has_country_flag = VIE_bubble_hit_hard
	}
}
