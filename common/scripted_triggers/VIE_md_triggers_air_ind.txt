	check_variable = { VIE_var_apm_active < 2 }
}

# ---------------------------------------------------------------------
# Tru 1 - A32 (dai tu khung may bay)
# ---------------------------------------------------------------------
VIE_apm_a32_t1_ok = {
	has_completed_focus = VIE_apm_a32
}
VIE_apm_a32_t2_ok = {
	check_variable = { VIE_apm_a32_tier > 0 }
}
VIE_apm_a32_t3_ok = {
	check_variable = { VIE_apm_a32_tier > 1 }
	OR = {
		AND = { has_country_flag = VIE_apm_a32_su30_life check_variable = { VIE_var_air_delivered > 11 } }
		check_variable = { VIE_ap_c295_qty > 0 }
	}
}
VIE_apm_a32_ok = {
	OR = {
		AND = { check_variable = { VIE_apm_a32_tier = 0 } VIE_apm_a32_t1_ok = yes }
		AND = { check_variable = { VIE_apm_a32_tier = 1 } VIE_apm_a32_t2_ok = yes }
		AND = { check_variable = { VIE_apm_a32_tier = 2 } VIE_apm_a32_t3_ok = yes }
	}
}
