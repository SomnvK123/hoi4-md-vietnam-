# ---------------------------------------------------------------------
# Cho trong chuong trinh: toi da 2 chay cung luc (VIE_var_apm_active, tu chua boi VIE_apm_slot_heal).
# ---------------------------------------------------------------------
VIE_apm_slot_free = {
	check_variable = { VIE_var_apm_active < 2 }
}

# ---------------------------------------------------------------------
# Tru 1 - A32 (dai tu khung may bay)
# ---------------------------------------------------------------------
VIE_apm_a32_t1_ok = {
	has_completed_focus = VIE_apm_a32
	date > 2010.12.31
}
VIE_apm_a32_t2_ok = {
	date > 2016.12.31
}
VIE_apm_a32_t3_ok = {
	date > 2023.12.31
	check_variable = { VIE_var_air_delivered > 11 }
	# it nhat mot huong phai chon duoc (event .13): Su-30MK2 (can co su30_life tu bac 2 option A) hoac C-295M (Truc 1 .8)
	OR = {
		has_country_flag = VIE_apm_a32_su30_life
		check_variable = { VIE_ap_c295_qty > 0 }
	}
}
VIE_apm_a32_ok = {
	OR = {
		AND = { check_variable = { VIE_apm_a32_tier = 0 } VIE_apm_a32_t1_ok = yes }
		AND = { check_variable = { VIE_apm_a32_tier = 1 } VIE_apm_a32_t2_ok = yes }
		AND = { check_variable = { VIE_apm_a32_tier = 2 } VIE_apm_a32_t3_ok = yes }
	}
}
