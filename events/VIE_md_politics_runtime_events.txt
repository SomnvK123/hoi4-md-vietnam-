country_event = {
	id = vie_pol_runtime.10
	title = vie_pol_runtime.10.t
	desc = vie_pol_runtime.10.d
	picture = GFX_report_event_generic_read_write
	is_triggered_only = yes
	trigger = { has_country_flag = VIE_pending_X01 NOT = { has_country_flag = VIE_record_X01 } }
	option = {
		name = vie_pol_runtime.10.a
		trigger = { NOT = { has_political_power < 20 }
		}
		if = {
			limit = { has_country_flag = VIE_pending_X01 NOT = { has_country_flag = VIE_record_X01 } NOT = { has_political_power < 20 }
			}
			if = {
				limit = { has_country_flag = VIE_pending_X01 NOT = { has_country_flag = VIE_record_X01 } NOT = { has_political_power < 20 }
				}
				add_political_power = -20
				set_country_flag = VIE_record_X01_A
				set_country_flag = VIE_record_X01
				clr_country_flag = VIE_pending_X01
				set_country_flag = VIE_gate_XI05
			}
		}
	}
	option = {
		name = vie_pol_runtime.10.b
		trigger = { NOT = { check_variable = { VIE_leadership_cohesion < 46 } }
		}
		if = {
			limit = { has_country_flag = VIE_pending_X01 NOT = { has_country_flag = VIE_record_X01 } NOT = { check_variable = { VIE_leadership_cohesion < 46 } }
			}
			if = {
				limit = { has_country_flag = VIE_pending_X01 NOT = { has_country_flag = VIE_record_X01 } NOT = { check_variable = { VIE_leadership_cohesion < 46 } }
				}
				set_temp_variable = { cohesion_add = -6 }
				VIE_modify_cohesion = yes
				set_country_flag = VIE_record_X01_B
				set_country_flag = VIE_record_X01
				clr_country_flag = VIE_pending_X01
				clr_country_flag = VIE_gate_XI05
			}
		}
	}
	option = {
		name = vie_pol_runtime.10.c
		trigger = { always = yes }
		if = {
			limit = { has_country_flag = VIE_pending_X01 NOT = { has_country_flag = VIE_record_X01 } always = yes }
		}
	}
}
