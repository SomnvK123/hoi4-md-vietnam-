	is_triggered_only = yes

	# Protest quietly
	option = {
		name = vie_scs.4.a

		add_opinion_modifier = { target = CHI modifier = VIE_hd981_restraint }
		reverse_add_opinion_modifier = { target = CHI modifier = VIE_hd981_restraint }
		ai_chance = { base = 70 }
	}

	# Hold live-fire exercises off the coast
	option = {
		name = vie_scs.4.b
		log = "[GetDateText]: [This.GetName]: Event vie_scs.4.b executed"
		set_temp_variable = { VIE_tension_add = 1 }
		VIE_add_scs_tension = yes
		navy_experience = 5
		add_war_support = 0.02
		add_opinion_modifier = { target = CHI modifier = VIE_hd981_defiance }
		reverse_add_opinion_modifier = { target = CHI modifier = VIE_hd981_defiance }
		ai_chance = {
			base = 30
			modifier = { factor = 0 VIE_ai_historical = yes }
			modifier = { factor = 3 VIE_ai_free = yes }
			modifier = { add = 60 has_global_flag = VIE_AI_PATH_NATIONALIST }
			modifier = { add = 40 has_global_flag = VIE_AI_PATH_AUTONOMY }
		}
	}
}
