########################################################################
# vie_axis.1: Developmental state captured by oligarchic interests
# Evans (1995): High market integration outpaces meritocratic capacity
########################################################################
country_event = {
	id = vie_axis.1
	title = vie_axis.1.t
	desc = vie_axis.1.d
	picture = GFX_report_event_generic_conference
	is_triggered_only = yes

	# Option A: Succumb to conglomerates and state capture
	option = {
		name = vie_axis.1.a
		log = "[GetDateText]: [This.GetName]: Event vie_axis.1.a executed"
		set_country_flag = VIE_oligarch_unlocked
		if = {
			limit = { NOT = { check_variable = { ruling_party = 15 } } }
			set_temp_variable = { party_index = 15 }
			set_temp_variable = { party_popularity_increase = 0.20 }
			set_temp_variable = { temp_outlook_increase = 0.20 }
			hidden_effect = { change_relative_party_popularity = yes }
			set_temp_variable = { rul_party_temp = 15 }
			VIE_transition_regime = yes
		}
		ai_chance = { base = 40 }
	}

	# Option B: Reassert discipline and crack down on financial collusion
	option = {
		name = vie_axis.1.b
		log = "[GetDateText]: [This.GetName]: Event vie_axis.1.b executed"
		add_political_power = -100
		add_stability = -0.05
		if = {
			limit = { has_idea = industrial_conglomerates }
			set_temp_variable = { temp_opinion = -20 }
			change_industrial_conglomerates_opinion = yes
		}
		add_to_variable = { VIE_ax_merit = 2 }
		VIE_ax_normalize = yes
		ai_chance = { base = 60 }
	}
}
