# E104: Hop tac ky thuat DNNN va an ninh du lieu (Caller: ngay 500)
# ------------------------------------------------------------
country_event = {

id = vie_pol.104
title = vie_pol.104.t
desc = vie_pol.104.d
picture = GFX_report_event_generic_read_write
is_triggered_only = yes
trigger = {
 has_country_flag = VIE_programme_active check_variable = { var = VIE_e104_offer_epoch value = VIE_programme_epoch compare = equals } NOT = { has_country_flag = VIE_e104_signed }
check_variable = { VIE_last_congress = 9 }
}
option = { name = vie_pol.104.a
trigger = {  NOT = { political_power < 30 }
NOT = { has_country_flag = VIE_e104_signed } has_country_flag = VIE_programme_active check_variable = { var = VIE_e104_offer_epoch value = VIE_programme_epoch compare = equals } OR = { NOT = { check_variable = { VIE_tier_cadre < 1 } } has_country_flag = VIE_dual_role_active } NOT = { has_country_flag = VIE_e104_signing } NOT = { has_country_flag = VIE_e104_offer_resolved }  check_variable = { VIE_last_congress = 9 } }
if = {
limit = { check_variable = { VIE_last_congress = 9 } }

if = {
limit = { NOT = { political_power < 30 }
NOT = { has_country_flag = VIE_e104_signed } has_country_flag = VIE_programme_active check_variable = { var = VIE_e104_offer_epoch value = VIE_programme_epoch compare = equals } OR = { NOT = { check_variable = { VIE_tier_cadre < 1 } } has_country_flag = VIE_dual_role_active } NOT = { has_country_flag = VIE_e104_signing } NOT = { has_country_flag = VIE_e104_offer_resolved } }
add_political_power = -30
set_variable = { VIE_e104_sign_days = 21 }
if = { limit = { has_country_flag = VIE_dual_role_active } set_variable = { VIE_e104_sign_days = 21 } }
else_if = { limit = { NOT = { check_variable = { VIE_tier_cadre < 3 } } } set_variable = { VIE_e104_sign_days = 21 } }
else_if = { limit = { check_variable = { VIE_tier_cadre = 2 } } set_variable = { VIE_e104_sign_days = 25 } }
else = { set_variable = { VIE_e104_sign_days = 30 } }
set_country_flag = VIE_e104_offer_resolved
set_variable = { VIE_e104_promised_support = 0.75 }
set_country_flag = VIE_record_E104_A
VIE_politics_e104_sign_begin = yes
}
}
}
option = { name = vie_pol.104.b
trigger = {  NOT = { political_power < 15 }
NOT = { has_country_flag = VIE_e104_signed } has_country_flag = VIE_programme_active check_variable = { var = VIE_e104_offer_epoch value = VIE_programme_epoch compare = equals } NOT = { has_country_flag = VIE_e104_signing } NOT = { has_country_flag = VIE_e104_offer_resolved }  check_variable = { VIE_last_congress = 9 } }
if = {
limit = { check_variable = { VIE_last_congress = 9 } }

if = {
limit = { NOT = { political_power < 15 }
NOT = { has_country_flag = VIE_e104_signed } has_country_flag = VIE_programme_active check_variable = { var = VIE_e104_offer_epoch value = VIE_programme_epoch compare = equals } NOT = { has_country_flag = VIE_e104_signing } NOT = { has_country_flag = VIE_e104_offer_resolved } }
add_political_power = -15
set_variable = { VIE_e104_sign_days = 21 }
set_country_flag = VIE_e104_offer_resolved
set_variable = { VIE_e104_promised_support = 0.25 }
set_country_flag = VIE_record_E104_B
VIE_politics_e104_sign_begin = yes
}
}
}
option = { name = vie_pol.104.c
trigger = { always = yes check_variable = { VIE_last_congress = 9 } }
if = {
limit = { check_variable = { VIE_last_congress = 9 } }

if = {
limit = { has_country_flag = VIE_programme_active NOT = { has_country_flag = VIE_e104_offer_resolved } check_variable = { var = VIE_e104_offer_epoch value = VIE_programme_epoch compare = equals } }
set_country_flag = VIE_e104_refused
set_country_flag = VIE_e104_offer_resolved}
}
}
