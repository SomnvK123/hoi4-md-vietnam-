	id = vie_hl.21
	title = vie_hl.21.t
	desc = vie_hl.21.d

	option = {
		name = vie_hl.21.a
		log = "[GetDateText]: [This.GetName]: Event vie_hl.21.a executed"
		if = {
			limit = { FROM = { tag = LAO } }
			set_temp_variable = { wargoal_on = LAO }
			add_threat_from_wargoal_effect = yes
			create_wargoal = { type = puppet_wargoal_focus target = LAO }
		}
		else_if = {
			limit = { FROM = { tag = CBD } }
			set_temp_variable = { wargoal_on = CBD }
			add_threat_from_wargoal_effect = yes
			create_wargoal = { type = puppet_wargoal_focus target = CBD }
		}
		set_temp_variable = { VIE_rp_add = 4 }
		VIE_hl_add_pressure = yes
		custom_effect_tooltip = VIE_hl_rp_p4_tt
		ai_chance = { base = 70 }
	}

	# Back down
	option = {
		name = vie_hl.21.b
		log = "[GetDateText]: [This.GetName]: Event vie_hl.21.b executed"
		add_political_power = -50
		add_stability = -0.02
		ai_chance = { base = 30 }
	}
}
