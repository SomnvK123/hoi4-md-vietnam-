########################################################################
# vie_dip.16  Who supplies the new fighters?  (opened by VIE_fighter_replacement)
########################################################################
country_event = {
	id = vie_dip.16
	title = vie_dip.16.t
	desc = vie_dip.16.d
	picture = GFX_report_event_generic_sign_treaty2
	is_triggered_only = yes

	# Stay with Russia: Su-30SM. MD has no Su-35 variant, so the Su-30 airframe is used
	# (variant "Su-30" of SOV exists in the By Blood Alone block; without the DLC the generic AS_Fighter2 is the closest type).
	option = {
		name = vie_dip.16.a
		trigger = {
			country_exists = SOV
			NOT = { has_war_with = SOV }
		}
		log = "[GetDateText]: [This.GetName]: Event vie_dip.16.a executed"
		add_opinion_modifier = { target = SOV modifier = VIE_arms_deal }
		reverse_add_opinion_modifier = { target = SOV modifier = VIE_arms_deal }
		set_temp_variable = { treasury_change = -4 }
		modify_treasury_effect = yes
		air_experience = 10
		if = {
			limit = { has_dlc = "By Blood Alone" }
			add_equipment_to_stockpile = {
				type = medium_plane_airframe_2
				amount = 12
				producer = SOV
			}
		}
		else = {
			add_equipment_to_stockpile = {
				type = AS_Fighter2
				amount = 12
				producer = SOV
			}
		}
		ai_chance = { base = 70 }
	}

	# Buy Western fighters (Gripen, as SIA does), possible once the embargo has been lifted
	option = {
		name = vie_dip.16.b
		trigger = {
			has_country_flag = VIE_us_arms_open
			country_exists = SWE
		}
		log = "[GetDateText]: [This.GetName]: Event vie_dip.16.b executed"
		add_opinion_modifier = { target = USA modifier = VIE_arms_deal }
		reverse_add_opinion_modifier = { target = USA modifier = VIE_arms_deal }
		add_opinion_modifier = { target = SWE modifier = VIE_arms_deal }
		reverse_add_opinion_modifier = { target = SWE modifier = VIE_arms_deal }
		set_temp_variable = { treasury_change = -5 }
		modify_treasury_effect = yes
		air_experience = 10
		add_equipment_to_stockpile = {
			type = MR_Fighter3
			amount = 12
			producer = SWE
		}
		set_temp_variable = { VIE_tension_add = 1 }
		VIE_add_scs_tension = yes
		ai_chance = {
			base = 30
			modifier = { factor = 0 VIE_ai_historical = yes }
		}
	}

	# Put off the decision, or build a home-grown answer once the defence industry can carry it
	option = {
		name = vie_dip.16.c
		log = "[GetDateText]: [This.GetName]: Event vie_dip.16.c executed"
		if = {
			limit = { has_completed_focus = VIE_modernize_vpa }
			air_experience = 25
			add_tech_bonus = {
				name = VIE_home_fighter
				category = CAT_medium_aircraft
				uses = 1
				bonus = 1.0
			}
		}
		else = {
			add_political_power = 25
		}
		ai_chance = { base = 10 }
	}
}
