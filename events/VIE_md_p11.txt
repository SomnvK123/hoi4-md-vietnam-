########################################################################
# vie_scs.16  Level 3: a clash at the Paracels (scheduler p11, once)
########################################################################
country_event = {
	id = vie_scs.16
	title = vie_scs.16.t
	desc = vie_scs.16.d
	picture = GFX_report_event_generic_battle
	is_triggered_only = yes

	# Show restraint
	option = {
		name = vie_scs.16.a
		log = "[GetDateText]: [This.GetName]: Event vie_scs.16.a executed"
		add_political_power = -75
		set_temp_variable = { VIE_tension_add = -1 }
		VIE_add_scs_tension = yes
		ai_chance = { base = 90 }
	}

	# Answer force with force: a limited claim on the Paracels
	option = {
		name = vie_scs.16.b
		log = "[GetDateText]: [This.GetName]: Event vie_scs.16.b executed"
		trigger = {
			country_exists = CHI
			NOT = { has_war_with = CHI }
		}
		add_state_claim = 813
		set_temp_variable = { wargoal_on = CHI }
		add_threat_from_wargoal_effect = yes
		create_wargoal = {
			type = take_state_focus
			target = CHI
			generator = { 813 }
		}
		add_war_support = 0.10
		navy_experience = 20
		set_country_flag = VIE_paracel_clash
		if = {
			limit = { has_completed_focus = VIE_limited_war_doctrine }
			set_country_flag = { flag = VIE_war_clock days = 60 }
		}
		else = {
			set_country_flag = { flag = VIE_war_clock days = 120 }
		}
		ai_chance = {
			base = 10
			modifier = { factor = 0 VIE_ai_historical = yes }
		}
	}
}
