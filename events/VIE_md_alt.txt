country_event = {
	id = vie_alt.1
	title = vie_alt.1.t

	picture = GFX_report_event_generic_conference
	is_triggered_only = yes

	# Historical: return to normal, with lessons learned
	option = {
		name = vie_alt.1.a
		log = "[GetDateText]: [This.GetName]: Event vie_alt.1.a executed"
		add_political_power = 25
		ai_chance = { base = 100 }
	}

	# The Party takes a hard line: strategic autonomy
	option = {
		name = vie_alt.1.b
		trigger = {
			check_variable = { VIE_hd981_response = 2 }
			VIE_scs_escalated_trigger = yes
			has_country_flag = VIE_bop_active
			power_balance_value = { id = VIE_party_balance value < 0.6 }
		}
		log = "[GetDateText]: [This.GetName]: Event vie_alt.1.b executed"
		VIE_bop_conservative_small = yes
		add_political_power = 25
		ai_chance = {
			base = 5
			modifier = { factor = 0 VIE_ai_historical = yes }
		}
	}
}

########################################################################
# vie_alt.14  Democratic path: the first free election
########################################################################
country_event = {
	id = vie_alt.14
	title = vie_alt.14.t
	desc = vie_alt.14.d
	picture = GFX_report_event_generic_rally
	is_triggered_only = yes

	# Liberals win
	option = {
		name = vie_alt.14.a
		log = "[GetDateText]: [This.GetName]: Event vie_alt.14.a executed"
		if = {
			limit = { NOT = { check_variable = { ruling_party = 2 } } }
			set_temp_variable = { party_index = 2 }
			set_temp_variable = { party_popularity_increase = 0.25 }
			set_temp_variable = { temp_outlook_increase = 0.25 }
			hidden_effect = { change_relative_party_popularity = yes }
			set_temp_variable = { rul_party_temp = 2 }
			VIE_transition_regime = yes
		}
		set_country_flag = VIE_democracy_elected
		country_event = { id = vie_int.6 days = 45 }
		ai_chance = { base = 50 }
	}

	# Conservatives win
	option = {
		name = vie_alt.14.b
		log = "[GetDateText]: [This.GetName]: Event vie_alt.14.b executed"
		if = {
			limit = { NOT = { check_variable = { ruling_party = 1 } } }
			set_temp_variable = { party_index = 1 }
			set_temp_variable = { party_popularity_increase = 0.25 }
			set_temp_variable = { temp_outlook_increase = 0.25 }
			hidden_effect = { change_relative_party_popularity = yes }
			set_temp_variable = { rul_party_temp = 1 }
			VIE_transition_regime = yes
		}
		set_country_flag = VIE_democracy_elected
		country_event = { id = vie_int.6 days = 45 }
		ai_chance = { base = 30 }
	}

	# Socialists win
	option = {
		name = vie_alt.14.c
		log = "[GetDateText]: [This.GetName]: Event vie_alt.14.c executed"
		if = {
			limit = { NOT = { check_variable = { ruling_party = 3 } } }
			set_temp_variable = { party_index = 3 }
			set_temp_variable = { party_popularity_increase = 0.25 }
			set_temp_variable = { temp_outlook_increase = 0.25 }
			hidden_effect = { change_relative_party_popularity = yes }
			set_temp_variable = { rul_party_temp = 3 }
			VIE_transition_regime = yes
		}
		set_country_flag = VIE_democracy_elected
		country_event = { id = vie_int.6 days = 45 }
		ai_chance = { base = 20 }
	}
}
