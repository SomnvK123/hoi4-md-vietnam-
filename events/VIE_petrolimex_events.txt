country_event = {
	id = vie_petro.1
	title = vie_petro.1.t
	desc = vie_petro.1.d
	picture = GFX_report_event_generic_factory

	is_triggered_only = no
	fire_only_once = yes

	trigger = {
		tag = VIE
		date > 2022.6.1
		NOT = { has_country_flag = VIE_petro_crisis_handled }
	}

	mean_time_to_happen = {
		months = 3
	}

	# Phương án A: Điều động Petrolimex mở bán 24/7 gánh toàn bộ thị trường
	option = {
		name = vie_petro.1.a
		trigger = {
			OR = {
				has_completed_focus = VIE_petrolimex_downstream_network
				has_completed_focus = VIE_strategic_petroleum_reserve
			}
		}
		log = "[GetDateText]: [This.GetName]: Event vie_petro.1.a executed"
		set_temp_variable = { treasury_change = -0.5 }
		modify_treasury_effect = yes
		add_stability = 0.02
		hidden_effect = { set_country_flag = VIE_petro_crisis_handled }
	}

	# Phương án B: Thả nổi theo biến động thế giới (Khủng hoảng thiếu nhiên liệu tạm thời)
	option = {
		name = vie_petro.1.b
		log = "[GetDateText]: [This.GetName]: Event vie_petro.1.b executed"
		add_timed_idea = { idea = VIE_fuel_shortage_crisis days = 365 }
		add_stability = -0.04
		hidden_effect = { set_country_flag = VIE_petro_crisis_handled }
	}
}
