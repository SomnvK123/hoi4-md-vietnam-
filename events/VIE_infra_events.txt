country_event = {
	id = vie_infra.5
	title = vie_infra.5.t
	desc = vie_infra.5.d
	picture = GFX_report_event_generic_factory
	is_triggered_only = yes
	option = {
		name = vie_infra.5.a
		trigger = { country_exists = JAP }
		log = "[GetDateText]: [This.GetName]: Event vie_infra.5.a executed"
		if = {
			limit = { NOT = { has_country_flag = VIE_infra_partner_resolved } }
			if = {
				limit = { VIE_infra_partner_unselected = yes country_exists = JAP }
				set_country_flag = VIE_hsr_partner_japan
				add_to_variable = { VIE_ax_west = 1 tooltip = VIE_ax_west_tt }
				VIE_ax_normalize = yes
				if = { limit = { country_exists = JAP } add_opinion_modifier = { target = JAP modifier = VIE_trade_agreement } reverse_add_opinion_modifier = { target = JAP modifier = VIE_trade_agreement } }
			}
			if = { limit = { VIE_infra_partner_selected = yes } set_country_flag = VIE_infra_partner_resolved clr_country_flag = VIE_infra_partner_pending }
			clr_country_flag = VIE_infra_partner_queued
		}
		ai_chance = { base = 50 }
	}
	option = {
		name = vie_infra.5.b
		trigger = { OR = { country_exists = GER country_exists = FRA } }
		log = "[GetDateText]: [This.GetName]: Event vie_infra.5.b executed"
		if = {
			limit = { NOT = { has_country_flag = VIE_infra_partner_resolved } }
			if = {
				limit = { VIE_infra_partner_unselected = yes OR = { country_exists = GER country_exists = FRA } }
				set_country_flag = VIE_hsr_partner_eu
				add_to_variable = { VIE_ax_west = 1 tooltip = VIE_ax_west_tt }
				add_to_variable = { VIE_ax_integ = 1 tooltip = VIE_ax_integ_tt }
				VIE_ax_normalize = yes
			}
			if = { limit = { VIE_infra_partner_selected = yes } set_country_flag = VIE_infra_partner_resolved clr_country_flag = VIE_infra_partner_pending }
			clr_country_flag = VIE_infra_partner_queued
		}
		ai_chance = { base = 50 }
	}
	option = {
		name = vie_infra.5.c
		trigger = { country_exists = CHI }
		log = "[GetDateText]: [This.GetName]: Event vie_infra.5.c executed"
		if = {
			limit = { NOT = { has_country_flag = VIE_infra_partner_resolved } }
			if = {
				limit = { VIE_infra_partner_unselected = yes country_exists = CHI }
				set_country_flag = VIE_hsr_partner_china
				add_to_variable = { VIE_ax_integ = 1 tooltip = VIE_ax_integ_tt }
				VIE_ax_normalize = yes
				if = { limit = { country_exists = CHI } add_opinion_modifier = { target = CHI modifier = VIE_trade_agreement } reverse_add_opinion_modifier = { target = CHI modifier = VIE_trade_agreement } }
			}
			if = { limit = { VIE_infra_partner_selected = yes } set_country_flag = VIE_infra_partner_resolved clr_country_flag = VIE_infra_partner_pending }
			clr_country_flag = VIE_infra_partner_queued
		}
		ai_chance = { base = 40 }
	}
	option = {
		name = vie_infra.5.d
		trigger = { NOT = { OR = { country_exists = JAP country_exists = CHI country_exists = GER country_exists = FRA } } }
		log = "[GetDateText]: [This.GetName]: Event vie_infra.5.d executed"
		clr_country_flag = VIE_infra_partner_queued
