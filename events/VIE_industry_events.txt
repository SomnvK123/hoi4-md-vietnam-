########################################################################
# vie_ind.2  Transshipment / origin-fraud scrutiny (only after the open-door choice)
########################################################################
country_event = {
	id = vie_ind.2
	title = vie_ind.2.t
	desc = vie_ind.2.d
	picture = GFX_report_event_generic_read_write
	is_triggered_only = yes

	# Crack down on origin fraud
	option = {
		name = vie_ind.2.a
		log = "[GetDateText]: [This.GetName]: Event vie_ind.2.a executed"
		set_temp_variable = { treasury_change = -1 }
		modify_treasury_effect = yes
		if = {
			limit = { country_exists = USA }
			add_opinion_modifier = { target = USA modifier = VIE_trade_agreement }
			reverse_add_opinion_modifier = { target = USA modifier = VIE_trade_agreement }
		}
		ai_chance = { base = 70 }
	}

	# Look the other way
	option = {
		name = vie_ind.2.b
		log = "[GetDateText]: [This.GetName]: Event vie_ind.2.b executed"
		add_timed_idea = { idea = VIE_origin_fraud_scrutiny days = 365 }
		ai_chance = { base = 30 }
	}
}
