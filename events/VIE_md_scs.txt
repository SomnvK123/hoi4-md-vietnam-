add_namespace = vie_scs

# South China Sea chain (Phase 1): HD-981 (May 2014) -> riots -> outcome.
# Variables/flags written here:
#   VIE_scs_tension (0-3)        raised by firm / internationalising responses; >=2 sets VIE_scs_escalated (unlocks E2 focuses)
#   VIE_hd981_response (1/2/3)   1 restraint, 2 firm, 3 internationalise
# CHI has its own scs events (china_scs.*) that may also target VIE; nothing here creates a war.

########################################################################
# vie_scs.1  HD-981 oil rig  -- turning point A
########################################################################
country_event = {
	id = vie_scs.1
	title = vie_scs.1.t
	desc = vie_scs.1.d
	picture = GFX_CHI_pla_navy
	is_triggered_only = yes

	immediate = {
		hidden_effect = { set_country_flag = VIE_hd981_active }
	}

	# Restraint: quiet diplomacy
	option = {
		name = vie_scs.1.a
		log = "[GetDateText]: [This.GetName]: Event vie_scs.1.a executed"
		set_variable = { VIE_hd981_response = 1 }
		VIE_bop_reform_small = yes
		add_opinion_modifier = { target = CHI modifier = VIE_hd981_restraint }
		reverse_add_opinion_modifier = { target = CHI modifier = VIE_hd981_restraint }
		add_timed_idea = { idea = VIE_public_anger days = 180 }
		country_event = { id = vie_scs.2 days = 10 random_days = 6 }
		ai_chance = {
			base = 50
		}
	}

	# Firm stance: coast guard and navy shadow the rig
	option = {
		name = vie_scs.1.b
		log = "[GetDateText]: [This.GetName]: Event vie_scs.1.b executed"
		set_variable = { VIE_hd981_response = 2 }
		set_temp_variable = { VIE_tension_add = 2 }
		VIE_add_scs_tension = yes
		VIE_bop_conservative_medium = yes
		add_political_power = -25
		add_war_support = 0.03
		navy_experience = 10
		add_opinion_modifier = { target = CHI modifier = VIE_hd981_defiance }
		reverse_add_opinion_modifier = { target = CHI modifier = VIE_hd981_defiance }
		country_event = { id = vie_scs.2 days = 10 random_days = 6 }
		ai_chance = {
			base = 20
			modifier = { factor = 0 VIE_ai_historical = yes }
		}
	}

	# Internationalise: press conferences, ASEAN, Washington and Tokyo
	option = {
		name = vie_scs.1.c
		log = "[GetDateText]: [This.GetName]: Event vie_scs.1.c executed"
		set_variable = { VIE_hd981_response = 3 }
		set_temp_variable = { VIE_tension_add = 1 }
		VIE_add_scs_tension = yes
		VIE_bop_reform_small = yes
		add_opinion_modifier = { target = USA modifier = VIE_hd981_appeal }
		reverse_add_opinion_modifier = { target = USA modifier = VIE_hd981_appeal }
		add_opinion_modifier = { target = JAP modifier = VIE_hd981_appeal }
		reverse_add_opinion_modifier = { target = JAP modifier = VIE_hd981_appeal }
		add_opinion_modifier = { target = CHI modifier = VIE_hd981_defiance }
		reverse_add_opinion_modifier = { target = CHI modifier = VIE_hd981_defiance }
		country_event = { id = vie_scs.2 days = 10 random_days = 6 }
		ai_chance = {
			base = 50
			modifier = { factor = 0.2 VIE_ai_historical = yes }
		}
	}
}

########################################################################
# vie_scs.2  Anti-China riots (mid May 2014)
########################################################################
country_event = {
	id = vie_scs.2
	title = vie_scs.2.t
	desc = vie_scs.2.d
	picture = GFX_report_event_generic_riot
	is_triggered_only = yes

	# Restore order (historical)
	option = {
		name = vie_scs.2.a
		log = "[GetDateText]: [This.GetName]: Event vie_scs.2.a executed"
		VIE_bop_conservative_small = yes
		add_stability = 0.02
		add_timed_idea = { idea = VIE_fdi_confidence_shock days = 120 }
		country_event = { id = vie_scs.3 days = 45 random_days = 10 }
		ai_chance = { base = 90 }
	}

	# Let the anger run its course
	option = {
		name = vie_scs.2.b
		log = "[GetDateText]: [This.GetName]: Event vie_scs.2.b executed"
		set_temp_variable = { VIE_tension_add = 1 }
		VIE_add_scs_tension = yes
		add_stability = -0.03
		add_war_support = 0.03
		add_timed_idea = { idea = VIE_fdi_confidence_shock days = 270 }
		country_event = { id = vie_scs.3 days = 45 random_days = 10 }
		ai_chance = {
			base = 10
			modifier = { factor = 0 VIE_ai_historical = yes }
		}
	}
}

########################################################################
# vie_scs.3  Rig withdrawn (mid July 2014) -- outcome depends on the stance taken
########################################################################
country_event = {
	id = vie_scs.3
	title = vie_scs.3.t
	desc = vie_scs.3.d
	picture = GFX_report_event_generic_conference
	is_triggered_only = yes

	immediate = {
		hidden_effect = { clr_country_flag = VIE_hd981_active }
		hidden_effect = { set_country_flag = VIE_hd981_done }
	}

	option = {
		name = vie_scs.3.a
		log = "[GetDateText]: [This.GetName]: Event vie_scs.3.a executed"
		if = {
			limit = { check_variable = { VIE_hd981_response = 1 } }
			add_political_power = 25
		}
		else_if = {
			limit = { check_variable = { VIE_hd981_response = 3 } }
			add_political_power = 50
		}
		else = {
			add_political_power = 25
			navy_experience = 10
		}
		if = {
			limit = { VIE_scs_escalated_trigger = yes }
			custom_effect_tooltip = VIE_e2_unlocked_tt
		}
	}
}
