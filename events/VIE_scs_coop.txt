# .1  Tap tran
country_event = {
	id = vie_scs_coop.1
	title = vie_scs_coop.1.t
	desc = vie_scs_coop.1.d
	picture = GFX_report_event_generic_read_write
	is_triggered_only = yes

	# A - Chi cac nuoc ASEAN: it XP, khong choc Trung Quoc
	option = {
		name = vie_scs_coop.1.a
		log = "[GetDateText]: [This.GetName]: Event vie_scs_coop.1.a executed"
		if = {
			limit = { has_selected_naval_grand_doctrine = yes }
			add_mastery = {
				amount = 10
				folder = naval
			}
		}
		else = {
			navy_experience = 10
		}
		if = {
			limit = { country_exists = PHI NOT = { has_war_with = PHI } }
			add_opinion_modifier = { target = PHI modifier = small_increase }
		}
		if = {
			limit = { country_exists = MAY NOT = { has_war_with = MAY } }
			add_opinion_modifier = { target = MAY modifier = small_increase }
		}
		if = {
			limit = { country_exists = IND NOT = { has_war_with = IND } }
			add_opinion_modifier = { target = IND modifier = small_increase }
		}
		if = {
			limit = { country_exists = SIN NOT = { has_war_with = SIN } }
			add_opinion_modifier = { target = SIN modifier = small_increase }
		}
		ai_chance = {
			base = 65
			modifier = { add = 100 VIE_ai_historical = yes }
		}
	}

	# B - Cung My / Nhat / Uc: nhieu XP, gan ket phuong Tay, Trung Quoc khong hai long
	option = {
		name = vie_scs_coop.1.b
		log = "[GetDateText]: [This.GetName]: Event vie_scs_coop.1.b executed"
		trigger = {
			OR = {
				country_exists = USA
				country_exists = JAP
				country_exists = AST
			}
		}
		if = {
			limit = { has_selected_naval_grand_doctrine = yes }
			add_mastery = {
				amount = 20
				folder = naval
			}
		}
		else = {
			navy_experience = 20
		}
		if = {
			limit = { country_exists = USA NOT = { has_war_with = USA } }
			add_opinion_modifier = { target = USA modifier = small_increase }
		}
		if = {
			limit = { country_exists = JAP NOT = { has_war_with = JAP } }
			add_opinion_modifier = { target = JAP modifier = small_increase }
		}
		if = {
			limit = { country_exists = AST NOT = { has_war_with = AST } }
			add_opinion_modifier = { target = AST modifier = small_increase }
		}
		if = {
			limit = { country_exists = CHI NOT = { has_war_with = CHI } }
			add_opinion_modifier = { target = CHI modifier = small_decrease }
		}
		ai_chance = {
			base = 35
			modifier = { factor = 0 has_active_mission = bankruptcy_incoming_collapse }
			modifier = { factor = 0 VIE_ai_historical = yes }
		}
	}
}
