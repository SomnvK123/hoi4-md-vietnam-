# vie_pol.1  Hidden init (fired once by VIE_event_scheduler)
#  - Party General Secretary Le Kha Phieu is the effective leader in 2000
#  - informal non-alignment policy, balance of power, South China Sea tension = 0
########################################################################
country_event = {
	id = vie_pol.1
	hidden = yes
	is_triggered_only = yes

	immediate = {
		log = "[GetDateText]: [This.GetName]: Event vie_pol.1 (init)"
		set_variable = { VIE_scs_tension = 0 }
		set_variable = { VIE_congress_term = 8 }
		add_ideas = VIE_non_alignment_policy
		if = {
			limit = { VIE_party_rule_active = yes }
			VIE_new_leader_phieu = yes
			set_power_balance = {
				id = VIE_party_balance
				left_side = VIE_conservative_side
				right_side = VIE_reform_side
				set_value = -0.1
			}
			set_country_flag = VIE_bop_active
		}
	}
}
