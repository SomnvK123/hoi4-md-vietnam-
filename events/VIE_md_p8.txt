########################################################################
# vie_int.10  Russia asks for Cam Ranh (opened by VIE_tc_arms_diversification)
########################################################################
country_event = {
	id = vie_int.10
	title = vie_int.10.t
	desc = vie_int.10.d
	picture = GFX_report_event_generic_sign_treaty2
	is_triggered_only = yes

	# Decline politely: no foreign bases
	option = {
		name = vie_int.10.a
		log = "[GetDateText]: [This.GetName]: Event vie_int.10.a executed"
		add_stability = 0.01
		add_political_power = 25
		ai_chance = { base = 80 }
	}

	# Accept, and give up the principle of no foreign bases
	option = {
		name = vie_int.10.b
		log = "[GetDateText]: [This.GetName]: Event vie_int.10.b executed"
		if = {
			limit = { has_idea = VIE_four_nos }
			remove_ideas = VIE_four_nos
		}
		if = {
			limit = { has_idea = VIE_non_alignment_policy }
			remove_ideas = VIE_non_alignment_policy
		}
		add_opinion_modifier = { target = SOV modifier = VIE_arms_deal }
		reverse_add_opinion_modifier = { target = SOV modifier = VIE_arms_deal }
		set_temp_variable = { treasury_change = 3 }
		modify_treasury_effect = yes
		VIE_bop_conservative_small = yes
		ai_chance = {
			base = 20
			modifier = { factor = 0 VIE_ai_historical = yes }
		}
	}
}
