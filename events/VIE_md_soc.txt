add_namespace = vie_soc

# Disasters and epidemics. Damage is lower if the matching preparedness focus was completed
# (VIE_disaster_prepared, VIE_mekong_adapted).

########################################################################
# vie_soc.1  SARS (2003)  -- class B
########################################################################
country_event = {
	id = vie_soc.1
	title = vie_soc.1.t
	desc = vie_soc.1.d
	picture = GFX_report_event_generic_read_write
	is_triggered_only = yes

	# Isolate the hospital, screen the borders
	option = {
		name = vie_soc.1.a
		log = "[GetDateText]: [This.GetName]: Event vie_soc.1.a executed"
		set_temp_variable = { treasury_change = -1 }
		modify_treasury_effect = yes
		add_stability = 0.02
		add_political_power = 25
		ai_chance = { base = 90 }
	}

	# Keep the news quiet to protect business
	option = {
		name = vie_soc.1.b
		log = "[GetDateText]: [This.GetName]: Event vie_soc.1.b executed"
		add_timed_idea = { idea = VIE_sars_outbreak days = 180 }
		ai_chance = {
			base = 10
			modifier = { factor = 0 VIE_ai_historical = yes }
			modifier = { factor = 3 VIE_ai_free = yes }
		}
	}
}

########################################################################
# vie_soc.3  Central Vietnam floods (random, at most one every ~3 years)
########################################################################
country_event = {
	id = vie_soc.3
	title = vie_soc.3.t
	desc = vie_soc.3.d
	picture = GFX_report_event_generic_ruins
	is_triggered_only = yes

	# Emergency relief
	option = {
		name = vie_soc.3.a
		log = "[GetDateText]: [This.GetName]: Event vie_soc.3.a executed"
		if = {
			limit = { has_country_flag = VIE_disaster_prepared }
			add_stability = 0.01
		}
		else = {
			set_temp_variable = { treasury_change = -1 }
			modify_treasury_effect = yes
			add_stability = 0.01
		}
		ai_chance = { base = 80 }
	}

	# Rely on local resources
	option = {
		name = vie_soc.3.b
		log = "[GetDateText]: [This.GetName]: Event vie_soc.3.b executed"
		if = {
			limit = { has_country_flag = VIE_disaster_prepared }
			add_stability = -0.01
		}
		else = {
			add_timed_idea = { idea = VIE_flood_damage days = 270 }
		}
		ai_chance = { base = 20 }
	}
}

	desc = vie_soc.4.d
	picture = GFX_report_event_generic_ruins
	is_triggered_only = yes

	option = {
		name = vie_soc.4.a
		log = "[GetDateText]: [This.GetName]: Event vie_soc.4.a executed"

		set_temp_variable = { temp_opinion = 4 }
		change_farmers_opinion = yes
		ai_chance = { base = 70 }
	}

	# Let farmers adapt on their own
	option = {
		name = vie_soc.4.b

		}

			set_temp_variable = { temp_opinion = -6 }
			add_stability = -0.02
		}
		change_farmers_opinion = yes
		ai_chance = { base = 30 }
	}
}
