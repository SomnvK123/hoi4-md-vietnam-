add_namespace = vie_sb

# State-building decisions (VIE_sb_*): each opens one of these events. See VIE_statebuilding_4axis_design.md.

country_event = {
	id = vie_sb.1
	title = vie_sb.1.t
	desc = vie_sb.1.d
	picture = GFX_report_event_generic_conference
	is_triggered_only = yes

	option = {
		name = vie_sb.1.a
		log = "[GetDateText]: [This.GetName]: Event vie_sb.1.a executed"
		add_to_variable = { VIE_sb_A_dec = -1 tooltip = VIE_sb_A_tt }
		if = {
			limit = { has_idea = communist_cadres }
			set_temp_variable = { temp_opinion = 5 }
			change_communist_cadres_opinion = yes
		}
		add_stability = 0.02
		VIE_ax_normalize = yes
		ai_chance = { base = 40 }
	}
	option = {
		name = vie_sb.1.b
		log = "[GetDateText]: [This.GetName]: Event vie_sb.1.b executed"
		add_to_variable = { VIE_sb_A_dec = 1 tooltip = VIE_sb_A_tt }
		if = {
			limit = { has_idea = communist_cadres }
			set_temp_variable = { temp_opinion = -5 }
			change_communist_cadres_opinion = yes
		}
		if = {
			limit = { NOT = { has_idea = bureau_01 } }
			decrease_centralization = yes
			set_temp_variable = { desire_change = -0.5 }
			change_expected_bureaucracy_spending = yes
		}
		VIE_ax_normalize = yes
		ai_chance = { base = 60 }
	}
	option = {
		name = vie_sb.1.c
		trigger = {
			has_completed_focus = VIE_party_discipline
			NOT = { VIE_party_rule_active = yes }
			NOT = { has_country_flag = VIE_sb_anticorr_cd }
			NOT = {
				OR = {
					has_idea = corruption_level_01
					has_idea = corruption_level_02
					has_idea = corruption_level_03
				}
			}
		}
		log = "[GetDateText]: [This.GetName]: Event vie_sb.1.c executed"
		add_to_variable = { VIE_sb_A_dec = 2 tooltip = VIE_sb_A_tt }
		if = {
			limit = { has_idea = communist_cadres }
			set_temp_variable = { temp_opinion = -5 }
			change_communist_cadres_opinion = yes
		}
		decrease_corruption = yes
		add_stability = -0.02
		set_country_flag = { flag = VIE_sb_anticorr_cd days = 730 }
		VIE_ax_normalize = yes
		ai_chance = { base = 20 }
	}
	option = {
		name = vie_sb.1.d
		trigger = {
			has_completed_focus = VIE_clean_cadres
			NOT = { has_country_flag = VIE_sb_rotation_cd }
		}
		log = "[GetDateText]: [This.GetName]: Event vie_sb.1.d executed"
		add_to_variable = { VIE_sb_A_dec = 1 tooltip = VIE_sb_A_tt }
		add_to_variable = { VIE_sb_C_dec = 0.5 tooltip = VIE_sb_C_tt }
		decrease_corruption = yes
		add_timed_idea = { idea = VIE_reorg_disruption days = 180 }
		set_country_flag = { flag = VIE_sb_rotation_cd days = 540 }
		VIE_ax_normalize = yes
		ai_chance = { base = 20 }
	}
}
