########################################################################
# vie_eco.19  US-China trade war (2018-19)  -- class B
########################################################################
country_event = {
	id = vie_eco.19
	title = vie_eco.19.t
	desc = vie_eco.19.d
	picture = GFX_report_event_generic_factory
	is_triggered_only = yes

	# Welcome factories moving out of China
	option = {
		name = vie_eco.19.a
		log = "[GetDateText]: [This.GetName]: Event vie_eco.19.a executed"
		increase_economic_growth = yes
		set_country_flag = VIE_china_plus_one_open
		set_temp_variable = { temp_opinion = 3 }
		change_industrial_conglomerates_opinion = yes
		ai_chance = { base = 70 }
	}

	# Tighten checks on goods re-exported from China
	option = {
		name = vie_eco.19.b
		log = "[GetDateText]: [This.GetName]: Event vie_eco.19.b executed"
		add_opinion_modifier = { target = USA modifier = VIE_trade_agreement }
		reverse_add_opinion_modifier = { target = USA modifier = VIE_trade_agreement }
		VIE_bop_conservative_small = yes
		ai_chance = {
			base = 30
			modifier = { factor = 0 VIE_ai_historical = yes }
		}
	}
}
