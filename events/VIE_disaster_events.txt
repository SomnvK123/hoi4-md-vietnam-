add_namespace = vie_disaster

########################################################################
# vie_disaster.1  Siêu bão Biển Đông đổ bộ Miền Trung & Vịnh Bắc Bộ
########################################################################
country_event = {
	id = vie_disaster.1
	title = vie_disaster.1.t
	desc = vie_disaster.1.d
	picture = GFX_report_event_generic_ruins

	is_triggered_only = no
	fire_only_once = yes

	trigger = {
		tag = VIE
		date > 2013.9.1
		has_completed_focus = VIE_disaster_preparedness
	}

	mean_time_to_happen = {
		months = 6
	}

	# Phương án A: Đã chuẩn bị đầy đủ Cứu hộ và Viễn thám
	option = {
		name = vie_disaster.1.a
		trigger = {
			has_completed_focus = VIE_military_rescue_corps
			has_completed_focus = VIE_satellite_early_warning
		}
		log = "[GetDateText]: [This.GetName]: Event vie_disaster.1.a executed"
		add_stability = 0.03
		add_political_power = 25
		if = {
			limit = { has_idea = the_military }
			set_temp_variable = { temp_opinion = 5 }
			change_the_military_opinion = yes
		}
	}

	# Phương án B: Đã chuẩn bị một phần
	option = {
		name = vie_disaster.1.b
		trigger = {
			OR = {
				has_completed_focus = VIE_military_rescue_corps
				has_completed_focus = VIE_satellite_early_warning
			}
			NOT = {
				AND = {
					has_completed_focus = VIE_military_rescue_corps
					has_completed_focus = VIE_satellite_early_warning
				}
			}
		}
		log = "[GetDateText]: [This.GetName]: Event vie_disaster.1.b executed"
		set_temp_variable = { treasury_change = -0.5 }
		modify_treasury_effect = yes
		add_stability = -0.01
	}

	# Phương án C: Chưa chuẩn bị (Thảm họa thiên tai)
	option = {
		name = vie_disaster.1.c
		trigger = {
			NOT = { has_completed_focus = VIE_military_rescue_corps }
			NOT = { has_completed_focus = VIE_satellite_early_warning }
		}
		log = "[GetDateText]: [This.GetName]: Event vie_disaster.1.c executed"
		set_temp_variable = { treasury_change = -2.0 }
		modify_treasury_effect = yes
		add_stability = -0.05
	}
}

########################################################################
# vie_disaster.2  Đại hạn mặn và khủng hoảng nguồn nước Mê Kông (2016)
########################################################################
country_event = {
	id = vie_disaster.2
	title = vie_disaster.2.t
	desc = vie_disaster.2.d
	picture = GFX_report_event_generic_factory

	is_triggered_only = no
	fire_only_once = yes

	trigger = {
		tag = VIE
		date > 2016.2.1
		has_completed_focus = VIE_disaster_preparedness
		NOT = { has_completed_focus = VIE_sustainable_mekong_delta }
	}

	mean_time_to_happen = {
		days = 15
	}

	# Hướng A: Kích hoạt khẩn cấp công trình ngăn mặn
	option = {
		name = vie_disaster.2.a
		log = "[GetDateText]: [This.GetName]: Event vie_disaster.2.a executed"
		set_temp_variable = { treasury_change = -1.0 }
		modify_treasury_effect = yes
		add_timed_idea = { idea = VIE_salinity_emergency days = 365 }
	}

	# Hướng B: Kích hoạt mô hình NQ120 Thuận thiên
	option = {
		name = vie_disaster.2.b
		log = "[GetDateText]: [This.GetName]: Event vie_disaster.2.b executed"
		set_temp_variable = { temp_opinion = 5 }
		change_farmers_opinion = yes
		add_timed_idea = { idea = VIE_nature_transition days = 365 }
	}
}
