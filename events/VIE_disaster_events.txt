add_namespace = vie_disaster

########################################################################
# vie_disaster.1  Siêu bão Biển Đông đổ bộ Miền Trung & Vịnh Bắc Bộ
########################################################################
country_event = {
	id = vie_disaster.1
	title = vie_disaster.1.t
	desc = vie_disaster.1.d
	picture = GFX_report_event_generic_ruins

	is_triggered_only = no
	fire_only_once = yes

	trigger = {
		tag = VIE
		date > 2013.9.1
	}

	mean_time_to_happen = {
		months = 6
	}

	# Phương án A: Đã có Luật Phòng thủ dân sự (hệ thống chỉ huy và lực lượng cứu hộ)
	option = {
		name = vie_disaster.1.a
		trigger = {
			has_completed_focus = VIE_civil_defense_law_2023
		}
		log = "[GetDateText]: [This.GetName]: Event vie_disaster.1.a executed"
		add_stability = 0.03
		add_political_power = 25
		if = {
			limit = { has_idea = the_military }
			set_temp_variable = { temp_opinion = 5 }
			change_the_military_opinion = yes
		}
	}

	# Phương án B: Mới có Luật Phòng, chống thiên tai
	option = {
		name = vie_disaster.1.b
		trigger = {
			has_completed_focus = VIE_disaster_law_2013
			NOT = { has_completed_focus = VIE_civil_defense_law_2023 }
		}
		log = "[GetDateText]: [This.GetName]: Event vie_disaster.1.b executed"
		set_temp_variable = { treasury_change = -0.5 }
		modify_treasury_effect = yes
		add_stability = -0.01
	}

	# Phương án C: Chưa chuẩn bị (Thảm họa thiên tai)
	option = {
		name = vie_disaster.1.c
		trigger = {
			NOT = { has_completed_focus = VIE_disaster_law_2013 }
		}
		log = "[GetDateText]: [This.GetName]: Event vie_disaster.1.c executed"
		set_temp_variable = { treasury_change = -2.0 }
		modify_treasury_effect = yes
		add_stability = -0.05
	}
}
